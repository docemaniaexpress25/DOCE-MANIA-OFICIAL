   846→        // BLOCO 40: metodo real quando vier na resposta (fallback A_PRAZO =
   847→        // padrao do pedido pendente gravado pela API)
   848→        metodoPagamento: s.metodo_pagamento || 'A_PRAZO',

   850→        statusPagamento: 'PENDENTE',
   851→        itens: (data.itens || []).map((i: any) => ({ produtoId: i.produtoId, quantidade: i.quantidade, precoVenda: i.precoVenda })),
   852→        dataVencimento: s.data_vencimento ? new Date(s.data_vencimento) : undefined,
   853→        trocas: s.trocas || undefined,
   854→        tipoVenda: 'PRE_VENDA',
   855→        entregaStatus: 'PENDENTE',
   856→        // BLOCO 40: modo do atendimento (NFE|NFCE|CUPOM) — o cupom mostra o
   857→        // documento certo (">>> NF-e <<<") direto no "Pedido Registrado!"
   858→        modoNota: s.modo_nota || undefined,
   859→      };
   860→      // BLOCO 13: pedido pendente — exclusao dele usa a API da fila
   861→      pvPendentesRef.current.add(mapped.id);
   862→      // BLOCO 88 (RESERVA): o servidor ja subtraiu o estoque — espelha NA TELA
   863→      // na hora (sem esperar o refetch). O catalogo da pre-venda filtra saldo
   864→      // > 0: vendeu tudo, o produto some do PDV no ato — nao da pra vender
   865→      // 15 un com 5 em estoque.
   866→      try {
   867→        const qtdMap = new Map<string, number>();
   868→        for (const it of (data.itens || [])) {
   869→          const pid = String(it.produtoId);
   870→          qtdMap.set(pid, (qtdMap.get(pid) || 0) + (Number(it.quantidade) || 0));
   871→        }
   872→        if (qtdMap.size > 0) {
   873→          setProducts(prev => prev.map(p => {
   874→            const q = qtdMap.get(p.id);
   875→            return q ? { ...p, estoquePrincipal: (Number(p.estoquePrincipal) || 0) - q } : p;
   876→          }));
   877→        }
   878→      } catch { /* espelho best-effort: o refetch corrige */ }
   879→      // Aviso de estoque insuficiente NO ATO da reserva (nao bloqueia o pedido)
   880→      if (Array.isArray(d.avisos) && d.avisos.length > 0) {
   881→        const linhas = d.avisos.map((a: any) => `${a.produto}: pediu ${a.solicitado}, tinha ${a.disponivel}`).join('\n');
   882→        setAdminNotification(`Atencao — reservou SEM estoque completo (saldo pode ficar negativo):\n${linhas}`);
   883→      }
   884→      // NAO entra no estado "sales" (venda so existe apos a entrega confirmada).
   885→      // O cupom usa o pedido retornado direto.
   886→      // BLOCO 21: previsão da comissão de PRÉ-VENDA calculada pelo SERVIDOR
   887→      // (mesma regra que gera a comissão na entrega — % própria do produto).
   888→      const comissaoPrevista = Number(d.comissaoPrevista);
   889→      return { pedido: mapped, comissaoPrevista: isFinite(comissaoPrevista) && comissaoPrevista > 0 ? comissaoPrevista : undefined };
   890→    } catch (e: any) {
   891→      console.error(e);
   892→      haptics.error();
   893→      return { pedido: null, erro: 'Sem conexao com o servidor.\nVerifique sua internet e tente novamente.' };
   894→    }

   948→
