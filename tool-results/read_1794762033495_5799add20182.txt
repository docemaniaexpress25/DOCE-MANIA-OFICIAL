    31→ * NOTAS FISCAIS (Bloco 10) — emissao de NF-e / NFC-e na venda de pronta entrega.
    32→ *
    33→ * BLOCO 85: payload atualizado para a API Focus v2.0 (formato "flat" da
    34→ * referencia oficial doc.focusnfe.com.br — ago/2026). Mudancas que importam:
    35→ * - destinatario RASO (cnpj_destinatario, nome_destinatario...), sem objeto cliente
    36→ * - pagamento: PIX = tPag 17 (16 e deposito bancario!)
    37→ * - NFC-e: data_emissao SEMPRE "agora" (SEFAZ recusa dhEmi > 5 min)
    38→ * - NF-e: emissao no momento em que o usuario "bate" a nota (data de emissao
    39→ *   real), referencia da venda vai nas informacoes adicionais
    40→ *
    41→ * POST /api/notas   { acao: 'EMITIR', saleId, tipo?: 'NFE'|'NFCE' }
    42→ *                   { acao: 'CONSULTAR', saleId }
    43→ *                   { acao: 'CANCELAR', saleId, justificativa }   [ADMIN]
    44→ * GET  /api/notas?saleId=...  -> status da integracao Focus + nota da venda
    45→ *                                (sem chamada externa; usado pelo Cupom para
    46→ *                                avisar quando o env FOCUS_NFE_TOKEN falta)
    47→ *
    48→ * Regras:
    49→ * - Vendedor so mexe nas PROPRIAS vendas; admin em qualquer uma.
    50→ * - Cliente com CNPJ valido -> NF-e (modelo 55). Consumidor final -> NFC-e.
    51→ * - NF-e exige cadastro minimo (CNPJ + razao — Bloco 77, regra do dono).
    52→ * - Se as colunas do Bloco 10 ainda nao existirem no Supabase, responde 503
    53→ *   com instrucao clara (probe serverSchema, padrao do projeto).
    54→ */

    82→function notaPublica(s: Record<string, unknown>): NotaPublica {
    83→  return {
    84→    status: (s.nota_status as string) || 'NAO_EMITIDA',
    85→    numero: (s.nota_numero as string) || '',
    86→    ref: (s.nota_ref as string) || '',
    87→    pdfUrl: (s.nota_pdf_url as string) || '',
    88→    erro: (s.nota_erro as string) || '',
    89→    // BLOCO 135: o front precisa saber se o documento é NFC-e ou NF-e
    90→    // ("NF 17" virava NFC/NF-e errado no card quando a consulta não dizia o tipo)
    91→    tipo: (s.nota_tipo as string) || '',
    92→    // BLOCO 142: data de autorização da SEFAZ — o front usa para o prazo de cancelar
    93→    emitidaEm: (s.nota_emitida_em as string) || null,
    94→  };
    95→}

   112→export async function GET(req: NextRequest) {
   113→  const bridged = await devBridge(req);
   114→  if (bridged) return bridged;
   115→  const session = sessionFromRequest(req);
   116→  if (!session) return BAD(401, 'Sessao expirada. Faca login novamente.');
   117→
   118→  const saleId = req.nextUrl.searchParams.get('saleId') || '';
   119→  if (!saleId) return BAD(400, 'Parametro ausente (saleId).');
   120→
   121→  // Status da integracao (sem tocar a Focus) — o Cupom usa para avisar
   122→  // quando o env da Vercel esta faltando em vez de erro generico no clique.
   123→  const cfg = getFocusConfig();
   124→  const focus = {
   125→    configurado: isFocusConfigured(),
   126→    ambiente: cfg.ambiente,
   127→    emitenteCnpj: !!cfg.cnpj,
   128→  };
   129→
   130→  if (!(await hasNotaColumns())) {
   131→    return NextResponse.json({ ok: true, focus, nota: null });
   132→  }
   133→
   134→  const db = getServiceClient();
   135→  const { data: sale, error } = await db.from('sales').select(await colsNota()).eq('id', saleId).single();
   136→  if (error || !sale) return BAD(404, 'Venda nao encontrada.');
   137→  if (session.perfil !== 'ADMIN' && session.sub !== sale.vendedor_id) {
   138→    return BAD(403, 'Voce so pode ver a nota das suas proprias vendas.');
   139→  }
   140→  return NextResponse.json({ ok: true, focus, nota: notaPublica(sale as Record<string, unknown>) });
   141→}

   143→export async function POST(req: NextRequest) {
   144→  const bridged = await devBridge(req);
   145→  if (bridged) return bridged;
   146→  const session = sessionFromRequest(req);
   147→  if (!session) return BAD(401, 'Sessao expirada. Faca login novamente.');
   148→
   149→  const body = await req.json().catch(() => ({}));
   150→  const acao = body.acao as string;
   151→  const saleId = body.saleId as string;
   152→  if (!saleId || !acao) return BAD(400, 'Parametros ausentes (acao, saleId).');
   153→
   154→  if (!(await hasNotaColumns())) {
   155→    return BAD(503, 'Colunas de nota fiscal nao existem ainda no Supabase. Rode o SQL do Bloco 10 (docs/NOTAS-FISCAIS.md).');
   156→  }
   157→
   158→  const db = getServiceClient();
   159→
   160→  // Carrega a venda e valida permissao
   161→  const { data: sale, error: saleErr } = await db
   162→    .from('sales')
   163→    .select(await colsNota())
   164→    .eq('id', saleId)
   165→    .single();
   166→
   167→  if (saleErr || !sale) return BAD(404, 'Venda nao encontrada.');
   168→  if (session.perfil !== 'ADMIN' && session.sub !== sale.vendedor_id) {
   169→    return BAD(403, 'Voce so pode emitir nota para suas proprias vendas.');
   170→  }
   171→
   172→  // ---------- CONSULTAR ----------
   173→  if (acao === 'CONSULTAR') {
   174→    if (!sale.nota_ref) return BAD(422, 'Esta venda ainda nao tem nota emitida.');
   175→    if (!isFocusConfigured()) return BAD(503, 'Focus NFe nao configurada (env FOCUS_NFE_TOKEN no Vercel).');
   176→
   177→    const doc: FocusDoc = sale.nota_tipo === 'NFCE' ? 'nfce' : 'nfe';
   178→    const r = await consultarNota(doc, sale.nota_ref as string);
   179→    if (!r.ok || !r.data || 'erro' in r.data) {
   180→      return BAD(502, (r.data as { erro?: string }).erro || 'Falha ao consultar a Focus NFe.', { nota: notaPublica(sale as Record<string, unknown>) });
   181→    }
   182→    const res = interpretarResultado(r.data as Record<string, unknown>);
   183→    const patch: Record<string, unknown> = {
   184→      nota_status: res.autorizada ? 'AUTORIZADA' : res.cancelada ? 'CANCELADA' : res.motivo ? 'REJEITADA' : 'EMITINDO',
   185→      nota_numero: res.numero || sale.nota_numero || null,
   186→      nota_pdf_url: res.pdfUrl || sale.nota_pdf_url || null,
   187→      nota_xml_url: res.xmlUrl || sale.nota_xml_url || null,
   188→      nota_erro: res.autorizada ? null : res.motivo || null,
   189→    };
   190→    // Bloco 25: guarda a chave de acesso (44 dig) p/ atrelar a NF-e ao boleto Inter
   191→    if (res.chave && (await hasBoletoColumns())) patch.nota_chave = res.chave;
   192→    // BLOCO 142: primeira vez que a SEFAZ confirma a autorização, grava a
   193→    // data-hora oficial — é a BASE do prazo de cancelamento (NFC-e 30min/NF-e 24h)
   194→    if (res.autorizada && res.dataAutorizacao && (await hasNotaEmitidaEm()) && !(sale as Record<string, unknown>).nota_emitida_em) {
   195→      const dAuto = new Date(res.dataAutorizacao);
   196→      if (!isNaN(dAuto.getTime())) patch.nota_emitida_em = dAuto.toISOString();
   197→    }
   198→    await atualizarNota(saleId, patch);
   199→    return NextResponse.json({ ok: true, nota: { ...notaPublica(sale as Record<string, unknown>), ...res } });
   200→  }
   201→
   202→  // ---------- CANCELAR (ADMIN) ----------
   203→  if (acao === 'CANCELAR') {
   204→    if (session.perfil !== 'ADMIN') return BAD(403, 'Somente o admin pode cancelar notas.');
   205→    if (!sale.nota_ref) return BAD(422, 'Esta venda nao tem nota para cancelar.');
   206→    const just = (body.justificativa as string) || '';
   207→    if (just.trim().length < 15) return BAD(422, 'Justificativa obrigatoria (minimo 15 caracteres — regra da SEFAZ).');
   208→    if (!isFocusConfigured()) return BAD(503, 'Focus NFe nao configurada (env FOCUS_NFE_TOKEN no Vercel).');
   209→
   210→    const doc: FocusDoc = sale.nota_tipo === 'NFCE' ? 'nfce' : 'nfe';
   211→
   212→    // ---------------------------------------------------------------
   213→    // BLOCO 142 — GUARDA DO PRAZO SEFAZ (NFC-e 30min / NF-e 24h após a
   214→    // autorização). Passado o prazo a SEFAZ RECUSA o evento — o servidor
   215→    // recusa antes com mensagem clara. Usa a data-hora REAL de autorização:
   216→    // sales.nota_emitida_em; quando ela não existe (venda antiga) consulta
   217→    // a Focus UMA vez (data_autorizacao) e grava. Sem timestamp real o
   218→    // julgamento fica com a SEFAZ — nunca bloqueia por chute (a NF-e pode
   219→    // ter sido batida DIAS depois da venda).
   220→    // ---------------------------------------------------------------
   221→    let prazo: ReturnType<typeof prazoCancelamentoNota> | null = null;
   222→    if (String(sale.nota_status).toUpperCase() === 'AUTORIZADA') {
   223→      let baseAutorizacao = (await hasNotaEmitidaEm()) ? ((sale as Record<string, unknown>).nota_emitida_em as string) || null : null;
   224→      if (!baseAutorizacao && sale.nota_ref) {
   225→        try {
   226→          const c = await consultarNota(doc, sale.nota_ref as string);
   227→          if (c.ok && c.data && !('erro' in c.data)) {
   228→            const ci = interpretarResultado(c.data as Record<string, unknown>);
   229→            if (ci.autorizada && ci.dataAutorizacao) {
   230→              const dAuto = new Date(ci.dataAutorizacao);
   231→              if (!isNaN(dAuto.getTime())) {
   232→                baseAutorizacao = dAuto.toISOString();
   233→                if (await hasNotaEmitidaEm()) await atualizarNota(saleId, { nota_emitida_em: baseAutorizacao });
   234→              }
   235→            }
   236→          }
   237→        } catch { /* Focus indisponível -> a SEFAZ decide no cancelamento */ }
   238→      }
   239→      prazo = prazoCancelamentoNota({ tipo: sale.nota_tipo as string, status: 'AUTORIZADA', emitidaEm: baseAutorizacao });
   240→      if (prazo.expirado && baseAutorizacao) {
   241→        return BAD(
   242→          422,
   243→          `Cancelamento FORA DO PRAZO da SEFAZ — ${doc === 'nfce' ? 'NFC-e' : 'NF-e'} autorizada em ${fmtDataBR(baseAutorizacao)} e o prazo de ${prazo.prazoLabel} já venceu. O evento de cancelamento seria recusado.`,
   244→          { nota: notaPublica(sale as Record<string, unknown>), prazo: { expiraEm: prazo.expiraEm, expirado: true, prazoLabel: prazo.prazoLabel } },
   245→        );
   246→      }
   247→    }
   248→
   249→    const r = await cancelarNota(doc, sale.nota_ref as string, just);
   250→    if (!r.ok || !r.data || 'erro' in r.data) {
   251→      return BAD(502, (r.data as { erro?: string }).erro || 'Falha ao cancelar na Focus NFe.', { nota: notaPublica(sale as Record<string, unknown>) });
   252→    }
   253→    const res = interpretarResultado(r.data as Record<string, unknown>);
   254→    await atualizarNota(saleId, {
   255→      nota_status: res.cancelada ? 'CANCELADA' : sale.nota_status,
   256→      nota_erro: res.cancelada ? null : res.motivo || null,
   257→    });
   258→    // BLOCO 116 (doc reference/cancelar_nfce.md): NFC-e so pode ser
   259→    // cancelada em ate 30 MINUTOS apos a emissao (NF-e: 24h). O aviso vai
   260→    // na resposta para o admin saber por que a SEFAZ pode recusar.
   261→    return NextResponse.json({
   262→      ok: res.cancelada,
   263→      aviso:
   264→        doc === 'nfce'
   265→          ? 'NFC-e so pode ser cancelada em ate 30 minutos apos a emissao (regra da SEFAZ).'
   266→          : undefined,
   267→      prazo: prazo ? { expiraEm: prazo.expiraEm, prazoLabel: prazo.prazoLabel, textoRestante: prazo.textoRestante } : undefined,
   268→      nota: { ...notaPublica(sale as Record<string, unknown>), ...res },
   269→    });
   270→  }
   271→
   272→  // ---------- EMITIR ----------
   273→  if (acao !== 'EMITIR') return BAD(400, `Acao desconhecida: ${acao}`);
   274→
   275→  if (sale.nota_status === 'AUTORIZADA') {
   276→    return NextResponse.json({ ok: true, jaEmitida: true, nota: notaPublica(sale as Record<string, unknown>) });
   277→  }
   278→  if (!isFocusConfigured()) {
   279→    return BAD(503, 'Focus NFe nao configurada. Defina FOCUS_NFE_TOKEN no Vercel (docs/NOTAS-FISCAIS.md).');
   280→  }
   281→
   282→  const cfg = getFocusConfig();
   283→
   284→  // ------------------------------------------------------------------
   285→  // BLOCO 108 — MODO CUPOM (regra do dono): cliente que "quer somente
   286→  // cupom" recebe e paga o cupom normal; a NFC-e ANONIMA que alimenta
   287→  // a fiscal sai com o PREÇO CUPOM do produto (products.preco_cupom;
   288→  // 0 = cai no preço de custo). Liga/desliga nas Configurações ERP.
   289→  // Cliente marcado modo_cupom entra no modo AUTOMATICAMENTE.
   290→  // ------------------------------------------------------------------
   291→  const erp = await lerErpConfig();
   292→  const temPrecoCupom = await hasPrecoCupom();
   293→  const temClientModoCupom = await hasClientModoCupom();
   294→  const temEmitidaEm = await hasNotaEmitidaEm();
   295→  const emitentePayload = (await lerEmitentePayload()) as Partial<FocusNfePayloadV2>;
   296→
   297→  // Cliente da venda
   298→  const { data: client } = await db
   299→    .from('clients')
   300→    .select(`nome_fantasia, nome, cnpj, telefone, endereco, bairro, razao_social, inscricao_estadual, endereco_numero, endereco_cep, endereco_municipio, endereco_uf, email${temClientModoCupom ? ', modo_cupom' : ''}`)
   301→    .eq('id', sale.client_id)
   302→    .single();
   303→
   304→  if (!client) return BAD(422, 'Cliente da venda nao encontrado.');
   305→
   306→  const modoCupomPed = body.modoCupom === true;
   307→  const clientModoCupom = temClientModoCupom && (client as Record<string, unknown>).modo_cupom === true;
   308→  if (modoCupomPed && !erp.cupomDuploNfc) {
   309→    return BAD(422, 'Modo cupom DESLIGADO. Ligue em Configuracoes -> ERP Fiscal & Automacoes (cupomDuploNfc).');
   310→  }
   311→  const modoCupom = erp.cupomDuploNfc && (modoCupomPed || clientModoCupom);
   312→
   313→  // Tipo automatico: CNPJ valido -> NFe; caso contrario NFC-e (consumidor)
   314→  const cpfCnpj = validaCpfCnpj(client.cnpj);
   315→  const tipoEscolhido = (body.tipo as string) || '';
   316→  let doc: FocusDoc;
   317→  if (modoCupom) {
   318→    doc = 'nfce'; // BLOCO 108: modo cupom e SEMPRE NFC-e anonima (sem consumidor)
   319→  } else if (tipoEscolhido === 'NFE' || tipoEscolhido === 'NFCE') {
   320→    doc = tipoEscolhido === 'NFE' ? 'nfe' : 'nfce';
   321→  } else {
   322→    doc = cpfCnpj === 'CNPJ' ? 'nfe' : 'nfce';
   323→  }
   324→
   325→  // ------------------------------------------------------------------
   326→  // BLOCO 137 (regra do dono): NFC-e ANONIMA = PREÇO ESPECIAL. A NFC-e sai
   327→  // sem consumidor (CONSUMIDOR NAO IDENTIFICADO) quando nao tem CPF
   328→  // digitado E o cadastro nao tem documento valido — ou quando e o modo
   329→  // cupom (BLOCO 108). Nesses casos os itens vao no PREÇO CUPOM do produto
   330→  // (preco_cupom; 0 = custo; 0 = preço da venda) e o total da NFC e a soma
   331→  // desses valores. BUG do papel corrigido: a NFC-e "sem CPF" emitida pela
   332→  // tela 8 do PDV ou pela fila do admin saia no PREÇO CHEIO da venda porque
   333→  // so o modo cupom explicito precificava (a mensagem do XML mostrava
   334→  // "Venda ... registrada pelo app" em vez de "Venda cupom ...").
   335→  // NFC-e identificada e NF-e continuam no preço digitado pelo vendedor.
   336→  // ------------------------------------------------------------------
   337→  const uf = String(client.endereco_uf || '').toUpperCase().trim();
   338→  const municipio = String(client.endereco_municipio || '').trim();
   339→  const logradouro = String(client.endereco || '').trim();
   340→  const bairro = String(client.bairro || '').trim();
   341→  const razao = String(client.razao_social || client.nome_fantasia || '').trim();
   342→  const cpfNaNota = String(body.cpf || '').replace(/\D/g, '');
   343→  const nfceIdentificada = doc === 'nfce' && (!!cpfNaNota || (!modoCupom && !!razao && !!cpfCnpj));
   344→  const nfceAnonima = doc === 'nfce' && !nfceIdentificada;
   345→  const precoAnonimo = modoCupom || nfceAnonima;
   346→
   347→  // BLOCO 117 (doc reference/emitir_nfce.md — NFCeRequest required): a NFC-e
   348→  // EXIGE cnpj_emitente no payload (a NF-e cai no cadastro da conta). Sem
   349→  // CNPJ (nem na tela ERP Fiscal, nem no env), recusamos ANTES de marcar a
   350→  // venda EMITINDO — com instrucao clara, em vez de rejeicao criptica da Focus.
   351→  if (doc === 'nfce') {
   352→    const cnpjEmit =
   353→      String(erp.emitente.cnpj || '').replace(/\D/g, '') || String(cfg.cnpj || '').replace(/\D/g, '');
   354→    if (!cnpjEmit) {
   355→      return BAD(
   356→        503,
   357→        'A NFC-e exige o CNPJ do emitente. Cadastre em Configuracoes -> ERP Fiscal & Automacoes (Dados do emitente) ou defina FOCUS_NFE_CNPJ na Vercel.'
   358→      );
   359→    }
   360→  }
   361→
   362→  // Itens + codigos fiscais (colunas ncm/cest/cfop se o Bloco 10 ja rodou)
   363→  const fiscalCols = await hasProductFiscalColumns();
   364→  const { data: items } = await db
   365→    .from('sale_items')
   366→    .select('produto_id, quantidade, preco_venda')
   367→    .eq('sale_id', saleId);
   368→  if (!items || items.length === 0) return BAD(422, 'Venda sem itens — nao e possivel emitir nota.');
   369→
   370→  const prodIds = [...new Set(items.map((i: Record<string, unknown>) => i.produto_id as string))];
   371→  const prodCols = [
   372→    'id, nome',
   373→    fiscalCols ? 'ncm, cest, cfop, ean, unidade, origem' : '',
   374→    // BLOCO 108: preco_cupom (0 = cai no custo) + custo como fallback
   375→    temPrecoCupom ? 'preco_cupom' : '',
   376→    'preco_custo',
   377→  ].filter(Boolean).join(', ');
   378→  const { data: prodsData } = await db
   379→    .from('products')
   380→    .select(prodCols)
   381→    .in('id', prodIds);
   382→  const prods = (prodsData || []) as unknown as Record<string, unknown>[];
   383→  const prodMap = new Map(prods.map(p => [p.id as string, p]));
   384→
   385→  const payloadItems = items.map((i: Record<string, unknown>) => {
   386→    const p = (prodMap.get(i.produto_id as string) || {}) as Record<string, unknown>;
   387→    // BLOCO 108 (regra do dono): no modo cupom a NFC anônima sai com o
   388→    // PREÇO CUPOM do produto; sem preco_cupom (0), usa o PREÇO DE CUSTO;
   389→    // sem custo também (0), cai no preço de venda da venda.
   390→    const precoCupom = temPrecoCupom ? Number(p.preco_cupom) || 0 : 0;
   391→    const custo = Number(p.preco_custo) || 0;
   392→    const valorUnitCupom = precoCupom > 0 ? precoCupom : custo > 0 ? custo : Number(i.preco_venda) || 0;
   393→    return {
   394→      codigo: String(p.id || '').slice(0, 8),
   395→      descricao: String(p.nome || 'Produto'),
   396→      ncm: fiscalCols ? (p.ncm as string) || undefined : undefined,
   397→      cfop: fiscalCols ? (p.cfop as string) || undefined : undefined,
   398→      ean: fiscalCols ? (p.ean as string) || undefined : undefined,
   399→      unidade: fiscalCols ? ((p.unidade as string) || 'UN') : 'UN',
   400→      origem: fiscalCols ? (p.origem as string) || undefined : undefined,
   401→      quantidade: Number(i.quantidade) || 0,
   402→      valorUnitario: precoAnonimo ? valorUnitCupom : (Number(i.preco_venda) || 0),
   403→    };
   404→  });
   405→
   406→  // BLOCO 108/137: na NFC-e ANONIMA (modo cupom ou consumidor nao
   407→  // identificado) o TOTAL da NFC e a soma dos preços especiais
   408→  // (a venda real continua no valor que o cliente pagou — gravado no banco)
   409→  const valorTotalCupom = Math.round(payloadItems.reduce((acc, it) => acc + it.quantidade * it.valorUnitario, 0) * 100) / 100;
   410→  const valorTotal = precoAnonimo ? valorTotalCupom : (Number(sale.valor_total) || 0);
   411→
   412→  // Pagamento v2.0 — PIX = tPag 17 (armadilha: 16 e deposito bancario)
   413→  const tp = tpagDeMetodo(sale.metodo_pagamento as string);
   414→  const pagamento: FocusFormaPagamento = {
   415→    indicador_pagamento: tp.indicador,
   416→    forma_pagamento: tp.forma,
   417→    valor_pagamento: Math.round(valorTotal * 100) / 100,
   418→  };
   419→
   420→  // Data de emissao: AGORA (NFC-e recusa dhEmi com +5 min; a NF-e "batida"
   421→  // pelo botao tambem emite na hora — a data da venda vai nas informacoes)
   422→  const dataEmissao = new Date();
   423→  const vendaCurta = String(saleId).slice(0, 8).toUpperCase();
   424→  const dataVendaTxt = sale.data_venda
   425→    ? new Intl.DateTimeFormat('pt-BR', { timeZone: 'America/Sao_Paulo', day: '2-digit', month: '2-digit', year: 'numeric' }).format(new Date(sale.data_venda as string))
   426→    : '';
   427→  const informacoes = precoAnonimo
   428→    ? `Venda cupom ${vendaCurta} — NFC-e nao identificada${modoCupom ? ' (modo cupom)' : ''}. Emitida em ${carimbo()}.`
   429→    : `Venda ${vendaCurta}${dataVendaTxt ? ` de ${dataVendaTxt}` : ''} registrada pelo app Doce Mania. NF-e emitida em ${carimbo()}.`;
   430→
   431→  // Destinatario + payload por tipo de documento
   432→  let payload;
   433→  if (doc === 'nfe') {
   434→    // BLOCO 77 (regra do dono): a NF-e NAO e batida por este app (o dono
   435→    // emite no sistema externo dele) — aqui o app so CONTROLA a venda como
   436→    // NF. Exigencia minima: CNPJ valido (+ razao, que cai no nome fantasia).
   437→    // Endereco/bairro/municipio/UF seguem do cadastro best-effort e NAO
   438→    // bloqueiam mais o registro da venda como NF.
   439→    const faltando: string[] = [];
   440→    if (!cpfCnpj) faltando.push('CNPJ valido');
   441→    if (!razao) faltando.push('Razao social / nome fantasia');
   442→    if (faltando.length > 0) {
   443→      return BAD(422, `Cadastro incompleto para NF-e. Complete no cadastro do cliente: ${faltando.join(', ')}.`);
   444→    }
   445→    const destinatario: FocusDestinatarioV2 = {
   446→      nome: razao,
   447→      cpf_cnpj: String(client.cnpj).replace(/\D/g, ''),
   448→      inscricao_estadual: String(client.inscricao_estadual || '').trim() || undefined,
   449→      logradouro: logradouro || undefined,
   450→      numero: numeroDoEndereco(client.endereco_numero as string, logradouro),
   451→      bairro: bairro || undefined,
   452→      municipio: municipio || undefined,
   453→      uf: uf || undefined,
   454→      cep: String(client.endereco_cep || '').replace(/\D/g, '') || undefined,
   455→      email: String((client as Record<string, unknown>).email || '').trim() || undefined,
   456→      telefone: String(client.telefone || '').replace(/\D/g, '') || undefined,
   457→    };
   458→    payload = montarPayloadNfe({
   459→      dataEmissao,
   460→      natureza: cfg.natureza,
   461→      ufEmitente: cfg.ufEmitente,
   462→      destinatario,
   463→      itens: payloadItems,
   464→      valorTotalVenda: valorTotal,
   465→      pagamento,
   466→      informacoesAdicionais: informacoes,
   467→      emitente: emitentePayload,
   468→    });
   469→  } else {
   470→    // NFC-e (Bloco 12): escolha explicita do vendedor na tela de venda
   471→    // - CPF informado (e valido): NFC-e identificando o consumidor
   472→    // - Sem CPF: NFC-e ANONIMA (CONSUMIDOR NAO IDENTIFICADO) — BLOCO 137:
   473→    //   sai no PREÇO ESPECIAL (precoAnonimo ja decidido acima)
   474→    let destinatario: FocusDestinatarioV2 | undefined;
   475→    if (cpfNaNota) {
   476→      if (validaCpfCnpj(cpfNaNota) !== 'CPF') {
   477→        return BAD(422, 'CPF invalido (digitos verificadores nao conferem). Corrija o CPF ou deixe vazio para NFC-e sem identificacao.');
   478→      }
   479→      const nomeConsumidor = String(client.nome || client.nome_fantasia || '').trim();
   480→      destinatario = {
   481→        nome: (nomeConsumidor || 'CONSUMIDOR').slice(0, 60),
   482→        cpf_cnpj: cpfNaNota,
   483→        logradouro: logradouro || undefined,
   484→        numero: numeroDoEndereco(client.endereco_numero as string, logradouro),
   485→        bairro: bairro || undefined,
   486→        municipio: municipio || undefined,
   487→        uf: uf || cfg.ufEmitente,
   488→        cep: String(client.endereco_cep || '').replace(/\D/g, '') || undefined,
   489→      };
   490→    } else if (razao && cpfCnpj && !modoCupom) {
   491→      // BLOCO 128 (pedido do dono): a NFC-e — escolhida na tela OU automatica —
   492→      // identifica o destinatario pelo CADASTRO sempre que o cliente tem
   493→      // CPF/CNPJ valido, e leva o ENDERECO COMPLETO do cadastro
   494→      // (logradouro/numero/bairro/MUNICIPIO/UF + CEP). A SEFAZ nao aceita
   495→      // destinatario sem documento, entao cadastro sem CPF/CNPJ valido segue
   496→      // anonimo (como antes — NFC-e de consumidor de passagem).
   497→      destinatario = {
   498→        nome: razao.slice(0, 60),
   499→        cpf_cnpj: String(client.cnpj).replace(/\D/g, ''),
   500→        logradouro: logradouro || undefined,
   501→        numero: numeroDoEndereco(client.endereco_numero as string, logradouro),
   502→        bairro: bairro || undefined,
   503→        municipio: municipio || undefined,
   504→        uf: uf || cfg.ufEmitente,
   505→        cep: String(client.endereco_cep || '').replace(/\D/g, '') || undefined,
   506→      };
   507→    }
   508→    // Sem CPF digitado E sem documento valido no cadastro: NFC-e anonima.
   509→    payload = montarPayloadNfce({
   510→      dataEmissao,
   511→      natureza: undefined, // v2.0 NFC-e tem padrao proprio ("VENDA AO CONSUMIDOR")
   512→      ufEmitente: cfg.ufEmitente,
   513→      destinatario,
   514→      itens: payloadItems,
   515→      valorTotalVenda: valorTotal,
   516→      pagamento,
   517→      informacoesAdicionais: informacoes,
   518→      emitente: emitentePayload,
   519→    });
   520→  }
   521→
   522→  const ref = refForSale(saleId, nextTentativa(sale.nota_ref as string));
   523→
   524→  // Marca EMITINDO antes de chamar a Focus (rastreabilidade se cair a conexao)
   525→  await atualizarNota(saleId, { nota_status: 'EMITINDO', nota_tipo: doc === 'nfce' ? 'NFCE' : 'NFE', nota_ref: ref, nota_erro: null });
   526→
   527→  const r = await emitirNfe(doc, ref, payload);
   528→  if (!r.ok || !r.data || 'erro' in r.data) {
   529→    const erro = (r.data as { erro?: string }).erro || `HTTP ${r.status}`;
   530→    await atualizarNota(saleId, { nota_status: 'REJEITADA', nota_erro: erro });
   531→    return BAD(502, erro, { nota: { status: 'REJEITADA', ref } });
   532→  }
   533→
   534→  const res = interpretarResultado(r.data as Record<string, unknown>);
   535→  const patch: Record<string, unknown> = {
   536→    nota_status: res.autorizada ? 'AUTORIZADA' : res.processando ? 'EMITINDO' : 'REJEITADA',
   537→    nota_numero: res.numero || null,
   538→    nota_pdf_url: res.pdfUrl || null,
   539→    nota_xml_url: res.xmlUrl || null,
   540→    nota_erro: res.autorizada ? null : res.motivo || 'Aguardando SEFAZ',
   541→  };
   542→  // Bloco 25: guarda a chave de acesso (44 dig) p/ atrelar a NF-e ao boleto Inter
   543→  if (res.chave && (await hasBoletoColumns())) patch.nota_chave = res.chave;
   544→  // BLOCO 108: carimbo de emissao autorizada (historico completo do admin)
   545→  if (res.autorizada && temEmitidaEm) patch.nota_emitida_em = new Date().toISOString();
   546→  const { error: errPatch } = await db.from('sales').update(patch).eq('id', saleId);
   547→  if (errPatch && temEmitidaEm) {
   548→    // coluna nota_emitida_em ainda nao existe (SQL 108 pendente) — grava sem ela
   549→    delete patch.nota_emitida_em;
   550→    await atualizarNota(saleId, patch);
   551→  }
   552→
   553→  // -------------------------------------------------------------------
   554→  // BLOCO 141 (regra do dono — "salve dentro do nosso sistema"): quando a
   555→  // nota autoriza na MESMA requisicao (sincrona — NFC-e e boa parte das
   556→  // NF-e), PDF e XML sao arquivados no NOSSO storage em background
   557→  // (after). Assim o portal/estoque de arquivos nasce junto com a nota,
   558→  // sem depender do botao IMPORTAR. Best-effort: falha nao afeta a venda.
   559→  // -------------------------------------------------------------------
   560→  if (res.autorizada) {
   561→    after(async () => {
   562→      try {
   563→        await arquivarDaConsulta(doc, ref, r.data as unknown as Record<string, unknown>);
   564→        invalidaPresenca();
   565→      } catch (e) {
   566→        console.warn('[notas:emitir] arquivamento em background falhou (importador cobre):', (e as Error).message);
   567→      }
   568→    });
   569→  }
   570→
   571→  return NextResponse.json({
   572→    ok: res.autorizada || res.processando,
   573→    nota: {
   574→      status: patch.nota_status,
   575→      numero: res.numero,
   576→      ref,
   577→      pdfUrl: res.pdfUrl,
   578→      erro: patch.nota_erro,
   579→      // BLOCO 135: tipo do documento emitido — badge certo (NFC/NF-e) na hora
   580→      tipo: doc === 'nfce' ? 'NFCE' : 'NFE',
   581→      // BLOCO 82: dados p/ imprimir o mini DANFE 80mm logo apos a autorizacao
   582→      // (sem precisar de outra consulta) — o app usa na hora da venda
   583→      chave: res.chave || '',
   584→      serie: res.serie || '',
   585→      protocolo: res.protocolo || '',
   586→      dataAutorizacao: res.dataAutorizacao || '',
   587→      qrCode: res.qrCode || '',
   588→      emitente: res.emitente,
   589→    },
   590→  });
   591→}
