    88→async function spedyRequest<T = unknown>(
    89→  method: 'GET' | 'POST' | 'PUT' | 'PATCH',
    90→  path: string,
    91→  body?: unknown,
    92→  timeoutMs = 45_000
    93→): Promise<{ ok: boolean; status: number; data: T & Record<string, unknown>; retryAfterSeconds?: number }> {
    94→  const { apiKey, baseUrl } = getSpedyConfig();
    95→  if (!apiKey) return { ok: false, status: 0, data: { erro: 'SPEDY_API_KEY nao configurada.' } as T & Record<string, unknown> };
    96→  try {
    97→    const res = await fetch(`${baseUrl}${path}`, {
    98→      method,
    99→      headers: { 'X-Api-Key': apiKey, 'Content-Type': 'application/json' },
   100→      body: body !== undefined ? JSON.stringify(body) : undefined,
   101→      signal: AbortSignal.timeout(timeoutMs),
   102→      cache: 'no-store',
   103→    });
   104→    const text = await res.text();
   105→    let data: unknown;
   106→    try {
   107→      data = text ? JSON.parse(text) : {};
   108→    } catch {
   109→      data = { erro: `Resposta nao-JSON da Spedy: ${text.slice(0, 200)}` };
   110→    }
   111→    const out: { ok: boolean; status: number; data: T & Record<string, unknown>; retryAfterSeconds?: number } = {
   112→      ok: res.ok,
   113→      status: res.status,
   114→      data: data as T & Record<string, unknown>,
   115→    };
   116→    if (res.status === 429) {
   117→      // Docs: corpo traz retryAfterSeconds; header Retry-After (segundos)
   118→      const ra = Number((data as Record<string, unknown>)?.retryAfterSeconds);
   119→      const hdr = Number(res.headers.get('retry-after'));
   120→      out.retryAfterSeconds = Number.isFinite(ra) && ra > 0 ? ra : Number.isFinite(hdr) && hdr > 0 ? hdr : 60;
   121→    }
   122→    return out;
   123→  } catch (e: unknown) {
   124→    const msg = e instanceof Error ? e.message : String(e);
   125→    return { ok: false, status: 0, data: { erro: `Falha de rede com a Spedy: ${msg}` } as T & Record<string, unknown> };
   126→  }
   127→}

   131→  path: string,
   132→  timeoutMs = 60_000
   133→): Promise<{ ok: boolean; status: number; buffer?: Buffer; contentType?: string; erro?: string }> {
   134→  const { apiKey, baseUrl } = getSpedyConfig();
   135→  if (!apiKey) return { ok: false, status: 0, erro: 'SPEDY_API_KEY nao configurada.' };
   136→  try {
   137→    const res = await fetch(`${baseUrl}${path}`, {
   138→      headers: { 'X-Api-Key': apiKey },
   139→      signal: AbortSignal.timeout(timeoutMs),
   140→      cache: 'no-store',
   141→    });
   142→    if (!res.ok) {
   143→      const text = await res.text().catch(() => '');
   144→      return { ok: false, status: res.status, erro: `Spedy ${res.status}: ${text.slice(0, 300)}` };
   145→    }
   146→    const ab = await res.arrayBuffer();
   147→    return { ok: true, status: res.status, buffer: Buffer.from(ab), contentType: res.headers.get('content-type') || undefined };
   148→  } catch (e: unknown) {
   149→    const msg = e instanceof Error ? e.message : String(e);
   150→    return { ok: false, status: 0, erro: `Falha de download na Spedy: ${msg}` };

   188→export async function enviarCertificadoSpedy(opts: {
   189→  empresaId?: string;
   190→  arquivoBase64: string; // conteudo do .pfx em base64
   191→  nomeArquivo?: string;
   192→  senha: string;
   193→}) {

   207→      signal: AbortSignal.timeout(60_000),
   208→    });
   209→    const text = await res.text();
   210→    let data: unknown;
   211→    try { data = text ? JSON.parse(text) : {}; } catch { data = { erro: text.slice(0, 300) }; }
   212→    return { ok: res.ok, status: res.status, data: data as Record<string, unknown> };
   213→  } catch (e: unknown) {
   214→    const msg = e instanceof Error ? e.message : String(e);
   215→    return { ok: false, status: 0, data: { erro: `Falha enviando certificado: ${msg}` } };

   240→/** Metodo do APP -> tPag da Spedy. A_PRAZO na NF-e = billetBanking (boleto
   241→ *  bancario, tPag 15) — as parcelas/vencimentos vao no bloco duplicates. */
   242→export function metodoSpedy(metodo?: string | null): SpedyMetodoPagamento {
   243→  const m = String(metodo || '').toUpperCase();
   244→  if (m === 'PIX') return 'pix';
   245→  if (m === 'CHEQUE') return 'check';
   246→  if (m === 'A_PRAZO') return 'billetBanking';
   247→  if (m === 'CARTAO_CREDITO' || m === 'CREDITO') return 'creditCard';
   248→  if (m === 'CARTAO_DEBITO' || m === 'DEBITO') return 'debitCard';
   249→  return 'money';
   250→}

   265→/** Duplicatas de uma venda a prazo: n parcelas iguais (a ultima absorve os
   266→ *  centavos de arredondamento) com vencimentos mensais a partir da data. */
   267→export function montarDuplicatas(opts: {
   268→  numeroNota: string;        // numero da nota (ou ref da venda antes de autorizar)
   269→  valorTotal: number;
   270→  parcelas: number;
   271→  primeiroVencimento?: Date;
   272→}): SpedyDuplicata[] {
   273→  const n = Math.max(1, Math.min(12, Math.round(opts.parcelas)));
   274→  if (n <= 1) {
   275→    return [{
   276→      number: `${opts.numeroNota}/1`,
   277→      dueDate: (opts.primeiroVencimento ?? new Date()).toISOString(),
   278→      amount: Math.round(opts.valorTotal * 100) / 100,
   279→    }];
   280→  }
   281→  const centavos = Math.round(opts.valorTotal * 100);
   282→  const base = Math.floor(centavos / n);
   283→  const resto = centavos - base * n;
   284→  const primeiro = opts.primeiroVencimento ?? new Date();
   285→  const dups: SpedyDuplicata[] = [];
   286→  for (let i = 0; i < n; i++) {
   287→    const venc = new Date(primeiro.getTime());
   288→    venc.setMonth(venc.getMonth() + i);
   289→    const valor = (base + (i === n - 1 ? resto : 0)) / 100;
   290→    dups.push({ number: `${opts.numeroNota}/${i + 1}`, dueDate: venc.toISOString(), amount: valor });
   291→  }
   292→  return dups;
   293→}

   355→function montarItensSpedy(opts: {
   356→  itens: ItemVendaInput[];
   357→  valorTotalVenda: number;
   358→  ufDestino: string;
   359→  ufEmitente: string;
   360→}): { items: SpedyItemPayload[] } {
   361→  const descontoTotal = round2(Math.max(0, round2(opts.itens.reduce((a, it) => a + it.quantidade * it.valorUnitario, 0)) - opts.valorTotalVenda));
   362→  const items: SpedyItemPayload[] = opts.itens.map((it, i) => {
   363→    const bruto = round2(it.quantidade * it.valorUnitario);
   364→    const desconto = i === 0 ? descontoTotal : 0;
   365→    return {
   366→      code: (it.codigo || `P${i + 1}`).slice(0, 60),
   367→      description: normaliza(it.descricao, 120),
   368→      ncm: soDigitos(it.ncm) || '21069090', // guarda do modulo Focus (doce/salgado generico)
   369→      cfop: Number(soDigitos(it.cfop) || (opts.ufDestino === opts.ufEmitente ? '5102' : '6102')) || 5102,
   370→      unit: (it.unidade || 'UN').toUpperCase().slice(0, 6),
   371→      quantity: it.quantidade,
   372→      unitAmount: round2(it.valorUnitario),
   373→      // trio tributavel espelhado (docs: item inline precisa dos 3)
   374→      unitTax: (it.unidade || 'UN').toUpperCase().slice(0, 6),
   375→      quantityTax: it.quantidade,
   376→      unitTaxAmount: round2(it.valorUnitario),
   377→      totalAmount: round2(bruto - desconto),
   378→      taxes: {
   379→        icms: { origin: Number(soDigitos(it.origem) || '0') || 0, csosn: 102 }, // Simples Nacional s/ credito
   380→        pis: { cst: 49 },    // outras saidas — DAS
   381→        cofins: { cst: 49 },
   382→      },
   383→    };
   384→  });
   385→  return { items };
   386→}

   388→function normaliza(v: string, max: number): string {
   389→  return (v || '').replace(/\s+/g, ' ').trim().slice(0, max);

   392→function montarReceiver(d: SpedyDestinatarioPayload): SpedyDestinatarioPayload {
   393→  const doc = soDigitos(d.federalTaxNumber);
   394→  const out: SpedyDestinatarioPayload = { name: normaliza(d.name, 60) || 'CONSUMIDOR FINAL' };
   395→  if (doc.length === 11 || doc.length === 14) out.federalTaxNumber = doc;
   396→  if (d.email) out.email = d.email.trim().slice(0, 80);
   397→  if (d.address) {
   398→    out.address = {
   399→      street: normaliza(d.address.street, 60),
   400→      number: (d.address.number || 'S/N').slice(0, 10),
   401→      district: normaliza(d.address.district, 60) || 'CENTRO',
   402→      postalCode: soDigitos(d.address.postalCode).slice(0, 8),
   403→      city: { name: normaliza(d.address.city?.name, 60) || 'CIDADE', state: (d.address.city?.state || 'RS').toUpperCase().slice(0, 2) },
   404→    };
   405→  }
   406→  return out;
   407→}

   409→/** ISO com fuso de SP (igual ao modulo Focus — offset fixo -03:00). */
   410→function isoComFuso(d: Date): string {
   411→  const p = new Intl.DateTimeFormat('sv-SE', {
   412→    timeZone: 'America/Sao_Paulo', year: 'numeric', month: '2-digit', day: '2-digit',
   413→    hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   414→  }).format(d);
   415→  return `${p.replace(' ', 'T')}-03:00`;
   416→}

   418→/** Payload NF-e (modelo 55) — venda com boleto A PRAZO ganha billing+duplicates. */
   419→export function montarPayloadNfeSpedy(opts: {
   420→  integrationId: string;
   421→  dataEmissao: Date;
   422→  natureza?: string;
   423→  ufEmitente?: string;
   424→  destinatario: SpedyDestinatarioPayload;
   425→  itens: ItemVendaInput[];
   426→  valorTotalVenda: number;
   427→  metodo: string;
   428→  /** Venda a prazo: parcelas + 1o vencimento (boleto Inter amarrado pelo nDup) */
   429→  parcelas?: number;
   430→  primeiroVencimento?: Date;
   431→  informacoesAdicionais?: string;
   432→}): SpedyNfePayload {
   433→  const cfg = getSpedyConfig();
   434→  const ufEmit = opts.ufEmitente || cfg.ufEmitente;
   435→  const ufDestino = '';
   436→  const { items } = montarItensSpedy({ itens: opts.itens, valorTotalVenda: opts.valorTotalVenda, ufDestino, ufEmitente: ufEmit });
   437→  const aPrazo = String(opts.metodo || '').toUpperCase() === 'A_PRAZO';
   438→  const payload: SpedyNfePayload = {
   439→    integrationId: opts.integrationId,
   440→    effectiveDate: isoComFuso(opts.dataEmissao),
   441→    operationNature: opts.natureza || cfg.natureza,
   442→    isFinalCustomer: true,
   443→    receiver: montarReceiver(opts.destinatario),
   444→    payments: [{ method: metodoSpedy(opts.metodo), amount: round2(opts.valorTotalVenda) }],
   445→    items,
   446→  };
   447→  if (aPrazo) {
   448→    const parcelas = Math.max(1, Math.min(12, opts.parcelas || 1));
   449→    payload.duplicates = montarDuplicatas({
   450→      numeroNota: opts.integrationId.replace(/^VENDA-/i, 'NF'),
   451→      valorTotal: opts.valorTotalVenda,
   452→      parcelas,
   453→      primeiroVencimento: opts.primeiroVencimento,
   454→    });
   455→    payload.billing = {
   456→      number: opts.integrationId,
   457→      originalAmount: round2(opts.valorTotalVenda),
   458→      netAmount: round2(opts.valorTotalVenda),
   459→    };
   460→  }
   461→  if (opts.informacoesAdicionais) {
   462→    payload.additionalInformation = normaliza(opts.informacoesAdicionais, 500);
   463→  }
   464→  return payload;
   465→}

   530→/** Status da nota na Spedy -> status do app (sales.nota_status). */
   531→export function interpretarResultadoSpedy(nota: Record<string, unknown>) {
   532→  const status = String(nota.status || '').toLowerCase();
   533→  const detail = (nota.processingDetail || {}) as Record<string, unknown>;
   534→  const erro = String(detail.erro || detail.message || nota.erro || '');
   535→  return {
   536→    id: String(nota.id || ''),
   537→    // enqueued | authorized | rejected | canceled | denied | disabled...
   538→    autorizada: status === 'authorized' || status === 'autorizado',
   539→    cancelada: status === 'canceled' || status === 'cancelado',
   540→    processando: ['enqueued', 'processing', 'processando_autorizacao'].includes(status),
   541→    rejeitada: status === 'rejected' || status === 'denied',
   542→    statusBruto: status,
   543→    numero: String(nota.number || ''),
   544→    serie: String(nota.series || ''),
   545→    chave: soDigitos(String(nota.accessKey || '')).slice(-44),
   546→    motivo: erro,
   547→    environment: String(nota.environment || ''),
   548→  };
   549→}

   589→/**

   595→  const raw = String(ref || '').trim();

   597→  return { saleId: m ? m[1] : null };

   599→
   600→/**
   601→ * BLOCO 95: verificacao de assinatura do webhook da Spedy — padrao STANDARD
   602→ * WEBHOOKS (docs.spedy.com.br): cabecalhos webhook-id, webhook-timestamp e
   603→ * webhook-signature ("v1,<base64>" — uma ou mais durante rotacao de segredo).
   604→ *   assinatura = HMAC-SHA256(segredo, "<id>.<timestamp>.<corpo>") em base64
   605→ * O segredo vem de GET /v1/webhooks/secret (env SPEDY_WEBHOOK_SECRET — aceita
   606→ * com ou sem prefixo whsec_). Comparacao em tempo constante + janela de 5
   607→ * minutos contra replay. SEM segredo configurado recusa tudo (o endpoint
   608→ * fica exposto na internet — nao aceita nada nao assinado).
   609→ */
   610→export async function verificarAssinaturaWebhook(
   611→  headers: Headers,
   612→  corpoCru: string
   613→): Promise<{ valido: boolean; motivo?: string }> {
   614→  const secret = process.env.SPEDY_WEBHOOK_SECRET || '';
   615→  if (!secret) {
   616→    return { valido: false, motivo: 'SPEDY_WEBHOOK_SECRET nao configurado no servidor (Vercel).' };
   617→  }
   618→
   619→  const id = headers.get('webhook-id') || '';
   620→  const ts = headers.get('webhook-timestamp') || '';
   621→  const sigHeader = headers.get('webhook-signature') || '';
   622→  if (!id || !ts || !sigHeader) {
   623→    return { valido: false, motivo: 'Cabecalhos Standard Webhooks ausentes (webhook-id/timestamp/signature).' };
   624→  }
   625→
   626→  // Janela de tolerancia: 5 minutos — fora disso e replay, recusa.
   627→  const tsNum = Number(ts);
   628→  if (!Number.isFinite(tsNum)) return { valido: false, motivo: 'webhook-timestamp invalido.' };
   629→  const agora = Math.floor(Date.now() / 1000);
   630→  if (Math.abs(agora - tsNum) > 300) {
   631→    return { valido: false, motivo: 'webhook-timestamp fora da tolerancia de 5 min (possivel replay).' };
   632→  }
   633→
   634→  // Segredo: "whsec_<base64>" (Standard Webhooks) ou base64 cru.
   635→  const secretoLimpo = secret.replace(/^whsec_/, '');
   636→  const secretBytes = Buffer.from(secretoLimpo, 'base64');
   637→  const key = secretBytes.length > 0 ? secretBytes : Buffer.from(secretoLimpo, 'utf8');
   638→
   639→  const esperado = crypto.createHmac('sha256', key).update(`${id}.${ts}.${corpoCru}`, 'utf8').digest('base64');
   640→
   641→  // "v1,abc v1,def" (rotacao de segredo) — vale qualquer uma.
   642→  for (const parte of sigHeader.split(/[ ,]+/)) {
   643→    const sig = parte.trim();
   644→    if (!sig) continue;
   645→    const v1 = sig.startsWith('v1,') ? sig.slice(3) : sig;
   646→    const a = Buffer.from(v1);
   647→    const b = Buffer.from(esperado);
   648→    if (a.length === b.length && crypto.timingSafeEqual(a, b)) return { valido: true };
   649→  }
   650→  return { valido: false, motivo: 'Assinatura HMAC invalida.' };
   651→}
