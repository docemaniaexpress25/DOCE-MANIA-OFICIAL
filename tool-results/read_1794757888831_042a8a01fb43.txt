   117→async function focusRequest<T = FocusNotaResultado>(
   118→  method: 'GET' | 'POST' | 'DELETE',
   119→  path: string,
   120→  body?: unknown
   121→): Promise<{ ok: boolean; status: number; data: T | { erro: string } }> {
   122→  const { baseUrl } = getFocusConfig();
   123→  try {
   124→    const res = await fetch(`${baseUrl}${path}`, {
   125→      method,
   126→      headers: {
   127→        Authorization: authHeader(),
   128→        'Content-Type': 'application/json',
   129→      },
   130→      body: body ? JSON.stringify(body) : undefined,
   131→      signal: AbortSignal.timeout(45_000),
   132→    });
   133→    const text = await res.text();
   134→    let data: unknown;
   135→    try {
   136→      data = text ? JSON.parse(text) : {};
   137→    } catch {
   138→      data = { erro: `Resposta nao-JSON da Focus: ${text.slice(0, 200)}` };
   139→    }
   140→    return { ok: res.ok, status: res.status, data: data as T };
   141→  } catch (e: unknown) {
   142→    const msg = e instanceof Error ? e.message : String(e);
   143→    return { ok: false, status: 0, data: { erro: `Falha de rede com a Focus NFe: ${msg}` } };
   144→  }
   145→}

   176→/** Valida CPF (11 digitos) ou CNPJ (14 digitos) por digitos verificadores */
   177→export function validaCpfCnpj(v?: string | null): 'CPF' | 'CNPJ' | null {
   178→  const s = soDigitos(v);
   179→  if (s.length === 11) {
   180→    if (/^(\d)\1{10}$/.test(s)) return null;
   181→    let soma = 0;
   182→    for (let i = 0; i < 9; i++) soma += parseInt(s[i]) * (10 - i);
   183→    let d1 = (soma * 10) % 11;
   184→    if (d1 === 10) d1 = 0;
   185→    if (d1 !== parseInt(s[9])) return null;
   186→    soma = 0;
   187→    for (let i = 0; i < 10; i++) soma += parseInt(s[i]) * (11 - i);
   188→    let d2 = (soma * 10) % 11;
   189→    if (d2 === 10) d2 = 0;
   190→    return d2 === parseInt(s[10]) ? 'CPF' : null;
   191→  }
   192→  if (s.length === 14) {
   193→    if (/^(\d)\1{13}$/.test(s)) return null;
   194→    const calc = (len: number): number => {
   195→      let soma = 0;
   196→      let peso = len - 7;
   197→      for (let i = 0; i < len; i++) {
   198→        soma += parseInt(s[i]) * peso--;
   199→        if (peso < 2) peso = 9;
   200→      }
   201→      const r = soma % 11;
   202→      return r < 2 ? 0 : 11 - r;
   203→    };
   204→    return calc(12) === parseInt(s[12]) && calc(13) === parseInt(s[13]) ? 'CNPJ' : null;
   205→  }
   206→  return null;
   207→}

   214→export function tpagDeMetodo(metodo?: string | null): { forma: string; indicador: 0 | 1 } {
   215→  const m = String(metodo || '').toUpperCase();
   216→  if (m === 'PIX') return { forma: '17', indicador: 0 };
   217→  if (m === 'CHEQUE') return { forma: '02', indicador: 1 };
   218→  if (m === 'A_PRAZO') return { forma: '15', indicador: 1 }; // boleto bancario
   219→  if (m === 'CARTAO_CREDITO' || m === 'CREDITO') return { forma: '03', indicador: 0 };
   220→  if (m === 'CARTAO_DEBITO' || m === 'DEBITO') return { forma: '04', indicador: 0 };
   221→  return { forma: '01', indicador: 0 }; // DINHEIRO e padrao
   222→}

   361→/** ISO com fuso de SP ("2026-11-01T10:30:00-03:00") — a referencia v2.0 usa
   362→ *  data-hora com timezone (sem horario de verao no BR desde 2019, offset
   363→ *  fixo -03:00). toISOString() ("...Z") tambem e ISO, mas o exemplo oficial
   364→ *  da docs e com offset — mandamos igual ao exemplo. */
   365→function isoComFuso(d: Date): string {
   366→  const p = new Intl.DateTimeFormat('sv-SE', {
   367→    timeZone: 'America/Sao_Paulo', year: 'numeric', month: '2-digit', day: '2-digit',
   368→    hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   369→  }).format(d);
   370→  return `${p.replace(' ', 'T')}-03:00`;
   371→}

   405→/** Itens v2.0 — valores NUMERICOS (a API v2.0 nao aceita mais string). */
   406→function montarItensV2(opts: {
   407→  itens: ItemVendaInput[];
   408→  valorTotalVenda: number;
   409→  ufDestino: string;
   410→  ufEmitente: string;
   411→  doc: FocusDoc;
   412→}): { items: FocusItemV2[]; valorProdutos: number; descontoTotal: number } {
   413→  const itensTotal = round2(opts.itens.reduce((acc, it) => acc + it.quantidade * it.valorUnitario, 0));
   414→  // Desconto (troca/desconto geral) entra concentrado no primeiro item
   415→  const descontoTotal = round2(Math.max(0, itensTotal - opts.valorTotalVenda));
   416→
   417→  const items: FocusItemV2[] = opts.itens.map((it, i) => {
   418→    const bruto = round2(it.quantidade * it.valorUnitario);
   419→    const desconto = i === 0 ? descontoTotal : 0;
   420→    const item: FocusItemV2 = {
   421→      // NFe v2.0 pede numero_item integer; NFC-e v2.0 pede string
   422→      numero_item: (opts.doc === 'nfe' ? i + 1 : String(i + 1)) as unknown as number,
   423→      codigo_produto: (it.codigo || `P${i + 1}`).slice(0, 60),
   424→      descricao: normalizaTexto(it.descricao, 120),
   425→      codigo_ncm: soDigitos(it.ncm) || '21069090',
   426→      cfop: soDigitos(it.cfop) || (opts.ufDestino === opts.ufEmitente ? '5102' : '6102'),
   427→      unidade_comercial: it.unidade || 'UN',
   428→      unidade_tributavel: it.unidade || 'UN',
   429→      quantidade_comercial: it.quantidade,
   430→      quantidade_tributavel: it.quantidade,
   431→      valor_unitario_comercial: round2(it.valorUnitario),
   432→      valor_unitario_tributavel: round2(it.valorUnitario),
   433→      valor_bruto: bruto,
   434→      inclui_no_total: 1,
   435→      icms_origem: Number(soDigitos(it.origem) || '0') || 0,
   436→      icms_situacao_tributaria: '102', // CSOSN 102 — Simples Nacional, sem ICMS
   437→      pis_situacao_tributaria: '49',   // Simples Nacional: outras saidas
   438→      cofins_situacao_tributaria: '49',
   439→    };
   440→    const ean = soDigitos(it.ean);
   441→    if (ean.length === 8 || ean.length === 12 || ean.length === 13 || ean.length === 14) {
   442→      item.codigo_barras_comercial = ean;
   443→      // BLOCO 130 (rejeicao "GTIN informado, mas nao informado o GTIN da unidade
   444→      // tributavel"): regra SEFAZ exige cEANT quando cEAN existe. O app sempre
   445→      // vende 1:1 (unidade/valor tributavel = comercial), entao o GTIN da
   446→      // unidade tributavel e o MESMO codigo de barras.
   447→      item.codigo_barras_tributavel = ean;
   448→    }
   449→    if (desconto > 0) item.valor_desconto = desconto;
   450→    return item;
   451→  });
   452→
   453→  return { items, valorProdutos: itensTotal, descontoTotal };
   454→}
   455→
   456→function destinatarioV2(d: FocusDestinatarioV2, doc: FocusDoc): Partial<FocusNfePayloadV2> {
   457→  const docDigitos = soDigitos(d.cpf_cnpj);
   458→  const out: Partial<FocusNfePayloadV2> = {};
   459→  if (d.nome) out.nome_destinatario = normalizaTexto(d.nome, 60);
   460→  if (docDigitos.length === 14) out.cnpj_destinatario = docDigitos;
   461→  else if (docDigitos.length === 11) out.cpf_destinatario = docDigitos;
   462→  const ie = (d.inscricao_estadual || '').trim();
   463→  if (ie && !/^isen/i.test(ie)) {
   464→    out.inscricao_estadual_destinatario = ie.slice(0, 20);
   465→    out.indicador_inscricao_estadual_destinatario = 1;
   466→  } else if (docDigitos) {
   467→    out.indicador_inscricao_estadual_destinatario = 9; // nao contribuinte
   468→  }
   469→  if (d.logradouro) out.logradouro_destinatario = normalizaTexto(d.logradouro, 60);
   470→  if (d.numero) out.numero_destinatario = d.numero.slice(0, 60);
   471→  if (d.bairro) out.bairro_destinatario = normalizaTexto(d.bairro, 60);
   472→  if (d.municipio) out.municipio_destinatario = normalizaTexto(d.municipio, 60);
   473→  if (d.uf) out.uf_destinatario = d.uf.toUpperCase().slice(0, 2);
   474→  if (soDigitos(d.cep)) out.cep_destinatario = soDigitos(d.cep);
   475→  if (d.email) out.email_destinatario = d.email.trim().slice(0, 60);
   476→  const tel = soDigitos(d.telefone);
   477→  if (tel.length >= 6) out.telefone_destinatario = tel;
   478→  return out;
   479→}

   631→/**
   632→ * Interpreta a resposta v2.0 da Focus num resultado normalizado do app.
   633→ * status possiveis: autorizado | processando_autorizacao | erro_autorizacao |
   634→ *                   cancelado | erro_cancelamento
   635→ */
   636→export function interpretarResultado(data: FocusNotaResultado) {
   637→  const status = (data.status || '').toLowerCase();
   638→  // Erros v2.0: erros[{mensagem, campo}] — mensagem legivel campo a campo
   639→  const errosV2 = (Array.isArray(data.erros) ? data.erros : [])
   640→    .map(e => [e?.campo, e?.mensagem].filter(Boolean).join(': '))
   641→    .filter(Boolean);
   642→  const isErro = status.startsWith('erro') || status === 'reprovado';
   643→  const motivo = errosV2.join('; ')
   644→    || (Array.isArray(data.motivos) ? data.motivos.join('; ') : data.motivos || '')
   645→    || data.erro_sefaz || ''
   646→    || (isErro ? (data.mensagem || data.mensagem_sefaz || data.erro || '') : (data.erro || ''));
   647→  const pdf = urlFocus(data.caminho_danfe) || urlFocus(data.danfe_url) || urlFocus(data.pdf_url) || urlFocus(data.url_danfe);
   648→  // BLOCO 82: dados para o mini DANFE 80mm (emitente real da SEFAZ, QR de
   649→  // consulta NFC-e quando a Focus devolve)
   650→  const emitenteNome = String(data.razao_social_emitente || data.nome_emitente || '').trim();
   651→  const emitenteCnpj = String(data.cnpj_emitente || '').replace(/\D/g, '');
   652→  return {
   653→    autorizada: status === 'autorizado',
   654→    cancelada: status.startsWith('cancelado'),
   655→    // FIX bloco 116 (doc v2.0): 'erro_autorizacao' e REJEICAO DEFINITIVA da
   656→    // SEFAZ (ex.: "Rejeição: Total da NF difere..."), NAO e processamento.
   657→    // Antes isso mapeava rejeitada como EMITINDO e a venda ficava presa.
   658→    processando: status === 'processando_autorizacao',
   659→    rejeitada: status === 'erro_autorizacao',
   660→    numero: data.numero || '',
   661→    cStatus: data.status_sefaz || '',
   662→    motivo,
   663→    pdfUrl: pdf,
   664→    xmlUrl: urlFocus(data.caminho_xml_nota_fiscal) || urlFocus(data.arquivo_url),
   665→    statusBruto: status,
   666→    // chave_nfe vem com prefixo ("NFe4119...") — fica so os 44 digitos
   667→    chave: soDigitos(data.chave_nfe || data.chave).slice(-44) || '',
   668→    serie: String(data.serie || ''),
   669→    protocolo: String(data.protocolo || ''),
   670→    dataAutorizacao: String(data.data_autorizacao || ''),
   671→    total: Number(data.total) || 0,
   672→    qrCode: String(data.qrcode_url || data.url_consulta_nf || data.qr_code || data.url_consulta || ''),
   673→    // Contingencia (doc emitir_nfce.md): NFC-e emitida offline nasce
   674→    // 'autorizado' com contingencia_offline=true e efetiva sozinha depois —
   675→    // o app avisa o usuario que o espelho/DANFE e valido mas o QR so
   676→    // consulta apos a efetivacao.
   677→    contingenciaOffline: data.contingencia_offline === true,
   678→    contingenciaEfetivada: data.contingencia_offline !== true ? null : data.contingencia_offline_efetivada !== false,
   679→    emitente: {
   680→      nome: emitenteNome,
   681→      cnpj: emitenteCnpj,
   682→      ie: String(data.ie_emitente || '').trim(),
   683→    },
   684→  };
   685→}
