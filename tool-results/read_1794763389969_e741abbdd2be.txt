   685→                              <button onClick={capturarGPSCupom} disabled={gpsSt === 'loading'} className="bg-amber-500 text-white text-[9px] font-black uppercase px-3 py-2.5 rounded-lg active:scale-95 disabled:opacity-50 flex-shrink-0">Capturar</button>
   686→                            )}
   687→                          </div>
   688→                          <button onClick={salvarContatoCupom} disabled={zapSaving} className="w-full bg-emerald-600 text-white font-black py-3 rounded-xl text-[9px] uppercase tracking-widest active:scale-95 disabled:opacity-60 flex items-center justify-center gap-1.5">
   689→                            <i className={`fa-solid ${zapSaving ? 'fa-spinner fa-spin' : 'fa-floppy-disk'}`}></i>{zapSaving ? 'Gravando...' : 'Salvar no banco'}
   690→                          </button>
   691→                        </div>
   692→                      )}
   693→                    </div>
   694→                  )}
   695→
   696→                  {allowDelete && onDeleteSale && (
   697→                    <button onClick={handleDelete} className="w-full bg-white text-rose-600 border border-rose-100 font-black py-3.5 rounded-2xl active:scale-95 text-[10px] uppercase flex items-center justify-center gap-2 hover:bg-rose-50 transition-colors">
   698→                      <i className="fa-solid fa-trash-can"></i> Estornar / Excluir Venda
   699→                    </button>
   700→                  )}
   701→                </div>
   702→              )}
   703→            </div>
   704→          )}
   705→          
   706→          <button onClick={onClose} className="w-full bg-slate-900 text-white font-black py-4 rounded-2xl active:scale-95 text-[10px] uppercase tracking-widest shadow-xl">
   707→            {closeLabel || "FECHAR"}
   708→          </button>
   709→        </div>
   710→      </div>
   711→          {modal && (
   712→        <ConfirmModal title={modal.title} message={modal.message} icon={modal.icon} iconColor={modal.iconColor} type={(modal.type as any) || 'confirm'} onConfirm={modal.onConfirm} onCancel={() => setModal(null)} />
   713→      )}
   714→      {/* BLOCO 134: JANELA FISCAL — NF-e / NFC-e identificada / NFC anônima,
   715→          cupom fiscal 80mm e situação na SEFAZ, tudo em UMA janela. A nota
   716→          atualizada volta para cá (banner pendente apaga ao autorizar).
   717→          BLOCO 136: a parte fiscal é o "número 2" — botão FISCAL abre a
   718→          janela com botão VOLTAR no cabeçalho. */}
   719→      {fiscalAberto && (
   720→        <JanelaFiscal
   721→          sale={sale}
   722→          client={clientLocal}
   723→          nota={nota}
   724→          focus={focus}
   725→          showToast={showToast}
   726→          onNotaChange={(n) => setNota(n)}
   727→          onClose={() => setFiscalAberto(false)}
   728→          podeCancelar={podeCancelarNota}
   729→          onNotaCancelada={onNotaCancelada}
   730→        />
   731→      )}
   732→
   733→      {/* ================================================================
   734→          BLOCO 136: JANELA DE IMPRESSÃO (o "número 1") — aberta pelo botão
   735→          IMPRIMIR CUPOM. Todo o "monte de informação" da imagem do dono
   736→          (canal Bluetooth x USB do PC, térmica, papel 56/80mm, imprimir,
   737→          copiar, WhatsApp) mora AQUI, com botão VOLTAR no cabeçalho.
   738→          Ao imprimir com sucesso a janela fecha sozinha (voltando ao cupom).
   739→          ================================================================ */}
   740→      {imprAberto && (
   741→        <div className="fixed inset-0 bg-black/90 z-[165] flex flex-col items-center p-4 pb-[calc(1rem+env(safe-area-inset-bottom))] overflow-y-auto backdrop-blur-sm">
   742→          <div className="bg-white w-full max-w-[420px] shadow-2xl flex flex-col shrink-0 rounded-t-3xl overflow-hidden my-auto animate-in zoom-in-95 duration-300">
   743→            {/* HEADER com VOLTAR */}
   744→            <div className="bg-blue-600 text-white px-4 py-3 flex items-center gap-2">
   745→              <button onClick={() => setImprAberto(false)} className="w-9 h-9 rounded-full bg-white/15 flex items-center justify-center active:scale-90" aria-label="Voltar">
   746→                <i className="fa-solid fa-arrow-left"></i>
   747→              </button>
   748→              <div className="flex-1 min-w-0">
   749→                <p className="text-[10px] font-black uppercase tracking-widest">Imprimir cupom</p>
   750→                <p className="text-[9px] font-bold text-white/70 truncate">{clientLocal?.nomeFantasia || clientLocal?.nome || 'Cliente'} · R$ {(Number(sale.valorTotal) || 0).toFixed(2)}</p>
   751→              </div>
   752→            </div>
   753→
   754→            <div className="p-4 space-y-3 bg-gray-100">
   755→              {/* BLOCO 139: IMAGEM DO CUPOM — o vendedor confere O QUE FOI PEDIDO emulando o papel impresso */}
   756→              {cupomImg && (
   757→                <div className="bg-white rounded-2xl border border-gray-200 shadow-sm p-3">
   758→                  <p className="text-[8px] font-black text-gray-400 uppercase tracking-widest mb-2 text-center"><i className="fa-solid fa-receipt mr-1"></i>Imagem do cupom — o que sai impresso</p>
   759→                  <img src={cupomImg} alt="Imagem do cupom emulado — como sai na térmica" className="w-full rounded-lg border border-gray-100" />
   760→                </div>
   761→              )}
   762→
   763→              {/* Status da impressora (movido do topo do cupom — BLOCO 136) */}
   764→              <div className={`px-3 py-2 rounded-xl flex items-center gap-2 text-[10px] font-bold ${badge.color}`}>
   765→                <i className={`fa-solid ${badge.icon}`}></i>
   766→                <span className="flex-1 break-words">{badge.text}</span>
   767→                {isConnected && printJob.status === 'idle' && (
   768→                  <button
   769→                    onClick={() => setModal({title:'Desconectar', message:'Desconectar a impressora?', icon:'fa-solid fa-bluetooth-b', type:'danger', onConfirm:()=>{setModal(null); bluetoothPrinter.forgetPrinter();}})}
   770→                    className="text-[9px] opacity-70 underline"
   771→                  >
   772→                    Desconectar
   773→                  </button>
   774→                )}
   775→              </div>
   776→
   777→              {/* Print progress bar */}
   778→              {printJob.status === 'printing' && (
   779→                <div className="h-1.5 bg-gray-200 rounded-full overflow-hidden">
   780→                  <div
   781→                    className="h-full bg-blue-600 rounded-full transition-all duration-300"
   782→                    style={{ width: `${printJob.progress || 0}%` }}
   783→                  />
   784→                </div>
   785→              )}
   786→
   787→              {/* BLOCO 111: canal de impressão — Bluetooth (celular) x USB do PC.
   788→                  A escolha fica gravada no aparelho e vale para todo o sistema. */}
   789→              <div className="flex bg-gray-200 p-1 rounded-2xl">
   790→                <button
   791→                  onClick={() => trocarCanal('BT')}
   792→                  className={`flex-1 py-2 rounded-xl text-[10px] font-black uppercase transition-all ${canalImpr === 'BT' ? 'bg-white shadow-sm text-blue-600' : 'text-gray-500'}`}
   793→                >
   794→                  <i className="fa-brands fa-bluetooth-b mr-1"></i>Bluetooth
   795→                </button>
   796→                <button
   797→                  onClick={() => trocarCanal('USB')}
   798→                  className={`flex-1 py-2 rounded-xl text-[10px] font-black uppercase transition-all ${canalImpr === 'USB' ? 'bg-white shadow-sm text-violet-600' : 'text-gray-500'}`}
   799→                >
   800→                  <i className="fa-solid fa-desktop mr-1"></i>USB do PC
   801→                </button>
   802→              </div>
   803→
   804→              {canalImpr === 'BT' ? (
   805→                <PrinterSelector accent="blue" />
   806→              ) : (
   807→                <p className="text-[9px] font-bold text-violet-600 bg-violet-50 border border-violet-100 rounded-xl px-3 py-2 leading-snug">
   808→                  <i className="fa-solid fa-circle-info mr-1"></i>A janela de impressão do computador vai abrir — escolha a térmica USB. O navegador memoriza a escolhida.
   809→                </p>
   810→              )}
   811→
   812→              {/* Paper width selector */}
   813→              <div className="flex bg-gray-200 p-1 rounded-2xl mb-1">
   814→                <button
   815→                  onClick={() => setPrintWidth('56MM')}
   816→                  className={`flex-1 py-2 rounded-xl text-[10px] font-black uppercase transition-all ${printWidth === '56MM' ? 'bg-white shadow-sm text-blue-600' : 'text-gray-500'}`}
   817→                >
   818→                  <i className="fa-solid fa-receipt mr-1"></i>56mm
   819→                </button>
   820→                <button
   821→                  onClick={() => setPrintWidth('80MM')}
   822→                  className={`flex-1 py-2 rounded-xl text-[10px] font-black uppercase transition-all ${printWidth === '80MM' ? 'bg-white shadow-sm text-blue-600' : 'text-gray-400'}`}
   823→                >
   824→                  <i className="fa-solid fa-receipt mr-1"></i>80mm
   825→                </button>
   826→              </div>
   827→
   828→              <div className="grid grid-cols-2 gap-2">
   829→                <button
   830→                  onClick={handlePrint}
   831→                  disabled={isPrinting}
   832→                  className={`font-black py-4 rounded-2xl flex items-center justify-center gap-2 active:scale-95 text-[10px] uppercase shadow-lg transition-all ${
   833→                    isPrinting
   834→                      ? 'bg-gray-300 text-gray-500 cursor-not-allowed'
   835→                      : canalImpr === 'USB' ? 'bg-violet-600 text-white' : 'bg-blue-600 text-white'
   836→                  }`}
   837→                >
   838→                  <i className={`fa-solid ${isPrinting ? 'fa-spinner fa-spin' : canalImpr === 'USB' ? 'fa-desktop' : 'fa-print'}`}></i>
   839→                  {isPrinting ? 'Imprimindo...' : canalImpr === 'USB' ? 'Imprimir (USB)' : 'Imprimir'}
   840→                </button>
   841→                <button onClick={handleCopy} className="bg-emerald-600 text-white font-black py-4 rounded-2xl flex items-center justify-center gap-2 active:scale-95 text-[10px] uppercase shadow-lg">
   842→                  <i className="fa-solid fa-copy"></i> Copiar
   843→                </button>
   844→              </div>
   845→
   846→              <button onClick={() => { setModal({title:'Enviar WhatsApp', message:'Deseja enviar este cupom pelo WhatsApp para o cliente?', icon:'fa-brands fa-whatsapp', iconColor:'text-white', onConfirm:()=>{setModal(null); handleWhatsApp();}}) }} className="w-full bg-green-50 text-green-600 border border-green-100 font-black py-2.5 rounded-2xl flex items-center justify-center gap-2 active:scale-95 text-[9px] uppercase transition-colors hover:bg-green-100">
   847→                <i className="fa-brands fa-whatsapp text-sm"></i> Enviar pelo WhatsApp
   848→              </button>
   849→
   850→              {/* ================================================================
   851→                  BLOCO 139: RECEBIMENTO REAL — depois do cupom não fiscal, ANTES
   852→                  de finalizar: "Essa venda foi à vista — você recebeu R$ X?
   853→                  [CONFIRMO] / [NÃO RECEBI]". CONFIRMO fecha o saldo no banco
   854→                  (caixa à vista); NÃO RECEBI deixa em aberto (Contas a Receber).
   855→                  ================================================================ */}
   856→              {mostraRecebimento && (
   857→                <div className="rounded-2xl border-2 border-emerald-300 bg-white p-3.5 space-y-2.5">
   858→                  <p className="text-[10px] font-black text-emerald-700 uppercase flex items-center gap-1.5"><i className="fa-solid fa-hand-holding-dollar"></i>Recebimento real</p>
   859→                  {respRecebimento === null && (
   860→                    <>
   861→                      <p className="text-[10px] font-black text-gray-700 leading-snug">Essa venda foi à vista — você recebeu os R$ {saldoCupom.toFixed(2)} da venda?</p>
   862→                      <div className="grid grid-cols-2 gap-2">
   863→                        <button onClick={confirmarRecebimentoCupom} className="py-3.5 rounded-xl bg-emerald-600 text-white text-[10px] font-black uppercase tracking-widest active:scale-95 shadow">Confirmo</button>
   864→                        <button onClick={naoRecebiCupom} className="py-3.5 rounded-xl bg-white border-2 border-rose-200 text-rose-600 text-[10px] font-black uppercase tracking-widest active:scale-95">Não recebi</button>
   865→                      </div>
   866→                      <p className="text-[8px] font-bold text-gray-400 leading-snug">NÃO RECEBI fica em Contas a Receber — cobra depois.</p>
   867→                    </>
   868→                  )}
   869→                  {respRecebimento === 'CONFIRMADO' && (
   870→                    <p className="text-[10px] font-black text-emerald-700 leading-snug bg-emerald-50 border border-emerald-100 rounded-xl px-3 py-2.5"><i className="fa-solid fa-circle-check mr-1"></i>Recebido — R$ {(Number(saleLocal.valorTotal) || 0).toFixed(2)} no caixa à vista.</p>
   871→                  )}
   872→                  {respRecebimento === 'NAO' && (
   873→                    <p className="text-[10px] font-black text-amber-700 leading-snug bg-amber-50 border border-amber-100 rounded-xl px-3 py-2.5"><i className="fa-solid fa-user-clock mr-1"></i>Registrado: fica em aberto — cobre depois no Contas a Receber.</p>
   874→                  )}
   875→                </div>
   876→              )}
   877→              {isVendaSalva && aPrazoCupom && saldoCupom > 0.005 && (
   878→                <p className="text-[9px] font-bold text-gray-500 bg-white border border-gray-200 rounded-xl px-3 py-2.5 leading-snug"><i className="fa-solid fa-user-clock mr-1 text-amber-500"></i>Venda a prazo — o recebimento real entra na data combinada (Contas a Receber).</p>
   879→              )}
   880→              {isVendaSalva && ePreVendaCupom && saldoCupom > 0.005 && (
   881→                <p className="text-[9px] font-bold text-gray-500 bg-white border border-gray-200 rounded-xl px-3 py-2.5 leading-snug"><i className="fa-solid fa-truck mr-1 text-emerald-500"></i>Pré-venda — o entregador recebe na entrega.</p>
   882→              )}
   883→
   884→              <button onClick={() => setImprAberto(false)} className="w-full py-3.5 text-gray-400 font-bold uppercase text-[10px] tracking-widest">Voltar ao cupom</button>
   885→            </div>
   886→          </div>
   887→        </div>
   888→      )}
   889→</div>
   890→  );
   891→};
   892→
   893→export default Cupom;
