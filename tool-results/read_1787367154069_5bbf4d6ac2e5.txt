    51→// Helpers para comprovante
    52→const compressImage = (file: File): Promise<string> => new Promise((resolve) => {
    53→  const reader = new FileReader();
    54→  reader.onload = (e) => {
    55→    const img = new Image();
    56→    img.onload = () => {
    57→      const canvas = document.createElement('canvas');
    58→      const MAX = 1200;
    59→      let w = img.width, h = img.height;
    60→      if (w > MAX || h > MAX) { if (w > h) { h = (h / w) * MAX; w = MAX; } else { w = (w / h) * MAX; h = MAX; } }
    61→      canvas.width = w; canvas.height = h;
    62→      canvas.getContext('2d')!.drawImage(img, 0, 0, w, h);
    63→      resolve(canvas.toDataURL('image/jpeg', 0.6));
    64→    };
    65→    img.src = e.target?.result as string;
    66→  };
    67→  reader.readAsDataURL(file);
    68→});
    69→
    70→const shareImage = async (base64: string, fileName: string) => {
    71→  const res = await fetch(base64);
    72→  const blob = await res.blob();
    73→  const file = new File([blob], fileName, { type: 'image/jpeg' });
    74→  if (navigator.share && navigator.canShare && navigator.canShare({ files: [file] })) {
    75→    await navigator.share({ files: [file], title: 'Comprovante de Venda' });
    76→  } else {
    77→    const url = URL.createObjectURL(blob);
    78→    const a = document.createElement('a');
    79→    a.href = url; a.download = fileName; a.click();
    80→    URL.revokeObjectURL(url);
    81→  }
