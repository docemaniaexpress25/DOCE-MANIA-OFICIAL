    89→/** Rótulo certo do documento: NFCE -> "NFC-e", NFE -> "NF-e" (regra BLOCO 135). */
    90→function rotuloTipo(tipo: string, modoNota: string): string {
    91→  const t = String(tipo || '').toUpperCase();
    92→  if (t === 'NFCE') return 'NFC-e';
    93→  if (t === 'NFE') return 'NF-e';
    94→  const m = String(modoNota || '').toUpperCase();
    95→  if (m === 'NFE') return 'NF-e';
    96→  if (m === 'NFCE') return 'NFC-e';
    97→  return 'NF';
    98→}

   103→const PERIODO_META: { key: FiltroPeriodo; label: string }[] = [
   104→  { key: 'HOJE', label: 'Hoje' },
   105→  { key: '7D', label: '7 dias' },
   106→  { key: '30D', label: '30 dias' },
   107→  { key: 'MES', label: 'Este mês' },

   110→
   111→function dentroPeriodo(dataVenda: string, p: FiltroPeriodo): boolean {
   112→  if (p === 'TUDO' || !dataVenda) return true;
   113→  const d = new Date(dataVenda);
   114→  if (!isFinite(d.getTime())) return true;
   115→  const agora = new Date();
   116→  const tz = 'America/Sao_Paulo';
   117→  const diaStr = (x: Date) => new Intl.DateTimeFormat('en-CA', { timeZone: tz, year: 'numeric', month: '2-digit', day: '2-digit' }).format(x);
   118→  if (p === 'HOJE') return diaStr(d) === diaStr(agora);
   119→  if (p === 'MES') return diaStr(d).slice(0, 7) === diaStr(agora).slice(0, 7);
   120→  const dias = p === '7D' ? 7 : 30;
   121→  return d.getTime() >= agora.getTime() - dias * 24 * 60 * 60 * 1000;
   122→}
