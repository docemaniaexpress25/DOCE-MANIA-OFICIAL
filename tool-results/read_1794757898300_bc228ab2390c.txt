   118→   118→  method: 'GET' | 'POST' | 'DELETE',
   119→   119→  path: string,

   129→   129→      },
   130→   130→      body: body ? JSON.stringify(body) : undefined,
   131→   131→      signal: AbortSignal.timeout(45_000),
   132→   132→    });
   133→   133→    const text = await res.text();
   134→   134→    let data: unknown;
   135→   135→    try {
   136→   136→      data = text ? JSON.parse(text) : {};
   137→   137→    } catch {
   138→   138→      data = { erro: `Resposta nao-JSON da Focus: ${text.slice(0, 200)}` };
   139→   139→    }
   140→   140→    return { ok: res.ok, status: res.status, data: data as T };
   141→   141→  } catch (e: unknown) {
   142→   142→    const msg = e instanceof Error ? e.message : String(e);
   143→   143→    return { ok: false, status: 0, data: { erro: `Falha de rede com a Focus NFe: ${msg}` } };

   145→   145→}

   176→   176→/** Valida CPF (11 digitos) ou CNPJ (14 digitos) por digitos verificadores */
   177→   177→export function validaCpfCnpj(v?: string | null): 'CPF' | 'CNPJ' | null {
   178→   178→  const s = soDigitos(v);
   179→   179→  if (s.length === 11) {
   180→   180→    if (/^(\d)\1{10}$/.test(s)) return null;
   181→   181→    let soma = 0;
   182→   182→    for (let i = 0; i < 9; i++) soma += parseInt(s[i]) * (10 - i);
   183→   183→    let d1 = (soma * 10) % 11;
   184→   184→    if (d1 === 10) d1 = 0;
   185→   185→    if (d1 !== parseInt(s[9])) return null;
   186→   186→    soma = 0;
   187→   187→    for (let i = 0; i < 10; i++) soma += parseInt(s[i]) * (11 - i);
   188→   188→    let d2 = (soma * 10) % 11;
   189→   189→    if (d2 === 10) d2 = 0;
   190→   190→    return d2 === parseInt(s[10]) ? 'CPF' : null;
   191→   191→  }
   192→   192→  if (s.length === 14) {
   193→   193→    if (/^(\d)\1{13}$/.test(s)) return null;
   194→   194→    const calc = (len: number): number => {
   195→   195→      let soma = 0;
   196→   196→      let peso = len - 7;
   197→   197→      for (let i = 0; i < len; i++) {
   198→   198→        soma += parseInt(s[i]) * peso--;
   199→   199→        if (peso < 2) peso = 9;
   200→   200→      }
   201→   201→      const r = soma % 11;
   202→   202→      return r < 2 ? 0 : 11 - r;
   203→   203→    };
   204→   204→    return calc(12) === parseInt(s[12]) && calc(13) === parseInt(s[13]) ? 'CNPJ' : null;
   205→   205→  }
   206→   206→  return null;
   207→   207→}

   406→   406→function montarItensV2(opts: {
   407→   407→  itens: ItemVendaInput[];
   408→   408→  valorTotalVenda: number;
   409→   409→  ufDestino: string;
   410→   410→  ufEmitente: string;
   411→   411→  doc: FocusDoc;
   412→   412→}): { items: FocusItemV2[]; valorProdutos: number; descontoTotal: number } {
   413→   413→  const itensTotal = round2(opts.itens.reduce((acc, it) => acc + it.quantidade * it.valorUnitario, 0));
   414→   414→  // Desconto (troca/desconto geral) entra concentrado no primeiro item
   415→   415→  const descontoTotal = round2(Math.max(0, itensTotal - opts.valorTotalVenda));
   416→   416→
   417→   417→  const items: FocusItemV2[] = opts.itens.map((it, i) => {
   418→   418→    const bruto = round2(it.quantidade * it.valorUnitario);
   419→   419→    const desconto = i === 0 ? descontoTotal : 0;
   420→   420→    const item: FocusItemV2 = {
   421→   421→      // NFe v2.0 pede numero_item integer; NFC-e v2.0 pede string
   422→   422→      numero_item: (opts.doc === 'nfe' ? i + 1 : String(i + 1)) as unknown as number,
   423→   423→      codigo_produto: (it.codigo || `P${i + 1}`).slice(0, 60),
   424→   424→      descricao: normalizaTexto(it.descricao, 120),
   425→   425→      codigo_ncm: soDigitos(it.ncm) || '21069090',
   426→   426→      cfop: soDigitos(it.cfop) || (opts.ufDestino === opts.ufEmitente ? '5102' : '6102'),
   427→   427→      unidade_comercial: it.unidade || 'UN',
   428→   428→      unidade_tributavel: it.unidade || 'UN',
   429→   429→      quantidade_comercial: it.quantidade,
   430→   430→      quantidade_tributavel: it.quantidade,
   431→   431→      valor_unitario_comercial: round2(it.valorUnitario),
   432→   432→      valor_unitario_tributavel: round2(it.valorUnitario),
   433→   433→      valor_bruto: bruto,
   434→   434→      inclui_no_total: 1,
   435→   435→      icms_origem: Number(soDigitos(it.origem) || '0') || 0,
   436→   436→      icms_situacao_tributaria: '102', // CSOSN 102 — Simples Nacional, sem ICMS
   437→   437→      pis_situacao_tributaria: '49',   // Simples Nacional: outras saidas
   438→   438→      cofins_situacao_tributaria: '49',
   439→   439→    };
   440→   440→    const ean = soDigitos(it.ean);
   441→   441→    if (ean.length === 8 || ean.length === 12 || ean.length === 13 || ean.length === 14) {
   442→   442→      item.codigo_barras_comercial = ean;
   443→   443→      // BLOCO 130 (rejeicao "GTIN informado, mas nao informado o GTIN da unidade
   444→   444→      // tributavel"): regra SEFAZ exige cEANT quando cEAN existe. O app sempre
   445→   445→      // vende 1:1 (unidade/valor tributavel = comercial), entao o GTIN da
   446→   446→      // unidade tributavel e o MESMO codigo de barras.
   447→   447→      item.codigo_barras_tributavel = ean;
   448→   448→    }
   449→   449→    if (desconto > 0) item.valor_desconto = desconto;
   450→   450→    return item;
   451→   451→  });
   452→   452→
   453→   453→  return { items, valorProdutos: itensTotal, descontoTotal };
   454→   454→}

   456→   456→function destinatarioV2(d: FocusDestinatarioV2, doc: FocusDoc): Partial<FocusNfePayloadV2> {
   457→   457→  const docDigitos = soDigitos(d.cpf_cnpj);

   461→   461→  else if (docDigitos.length === 11) out.cpf_destinatario = docDigitos;
   462→   462→  const ie = (d.inscricao_estadual || '').trim();
   463→   463→  if (ie && !/^isen/i.test(ie)) {
   464→   464→    out.inscricao_estadual_destinatario = ie.slice(0, 20);

   469→   469→  if (d.logradouro) out.logradouro_destinatario = normalizaTexto(d.logradouro, 60);
   470→   470→  if (d.numero) out.numero_destinatario = d.numero.slice(0, 60);

   472→   472→  if (d.municipio) out.municipio_destinatario = normalizaTexto(d.municipio, 60);
   473→   473→  if (d.uf) out.uf_destinatario = d.uf.toUpperCase().slice(0, 2);
   474→   474→  if (soDigitos(d.cep)) out.cep_destinatario = soDigitos(d.cep);
