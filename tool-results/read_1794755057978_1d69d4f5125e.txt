    69→/** Rotulo da nota (NF-e / NFC-e) conforme nota_tipo gravado na emissao (Bloco 10) */
    70→function notaRotulo(tipo: string | null | undefined, numero: string | null | undefined) {
    71→  const t = String(tipo || '').toUpperCase();
    72→  const prefix = t === 'NFE' ? 'NF-e' : t === 'NFCE' ? 'NFC-e' : 'NF';
    73→  const num = String(numero || '').trim();
    74→  return num ? `${prefix} nº ${num}` : prefix;
    75→}

    77→/** Comprime a foto do comprovante no proprio aparelho antes de enviar */
    78→function compressImage(file: File): Promise<string> {
    79→  return new Promise((resolve, reject) => {
    80→    const reader = new FileReader();
    81→    reader.onload = () => {
    82→      const img = new Image();
    83→      img.onload = () => {
    84→        const max = 1280;
    85→        let { width, height } = img;
    86→        if (width > max || height > max) {
    87→          const ratio = Math.min(max / width, max / height);
    88→          width = Math.round(width * ratio);
    89→          height = Math.round(height * ratio);
    90→        }
    91→        const canvas = document.createElement('canvas');
    92→        canvas.width = width;
    93→        canvas.height = height;
    94→        const ctx = canvas.getContext('2d');
    95→        if (!ctx) { reject(new Error('Canvas indisponivel')); return; }
    96→        ctx.drawImage(img, 0, 0, width, height);
    97→        resolve(canvas.toDataURL('image/jpeg', 0.75));
    98→      };
    99→      img.onerror = () => reject(new Error('Imagem invalida'));
   100→      img.src = reader.result as string;
   101→    };
   102→    reader.onerror = () => reject(new Error('Erro ao ler arquivo'));
   103→    reader.readAsDataURL(file);
   104→  });
   105→}

   107→/** Copia com fallback para WebView/APK antigos (sem Clipboard API) */
   108→async function copyText(text: string): Promise<boolean> {
   109→  try {
   110→    await navigator.clipboard.writeText(text);
   111→    return true;
   112→  } catch {
   113→    try {
   114→      const ta = document.createElement('textarea');
   115→      ta.value = text;
   116→      ta.style.position = 'fixed';
   117→      ta.style.opacity = '0';
   118→      document.body.appendChild(ta);
   119→      ta.focus();
   120→      ta.select();
   121→      const ok = document.execCommand('copy');
   122→      document.body.removeChild(ta);
   123→      return ok;
   124→    } catch {
   125→      return false;
   126→    }
   127→  }
   128→}
