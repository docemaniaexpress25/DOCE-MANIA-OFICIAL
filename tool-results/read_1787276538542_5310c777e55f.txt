    47→  
    48→  const [cart, setCart] = useState<Record<string, CartItem>>(() =>

    52→  const [metodo, setMetodo] = useState<PaymentMethod>('DINHEIRO');
    53→  const [detalheMetodo, setDetalheMetodo] = useState<string>('Dinheiro');
    54→  const [tipoPrazo, setTipoPrazo] = useState<'PRAZO_COMUM' | 'CHEQUE' | 'BOLETO'>('PRAZO_COMUM');
    55→  const [prazoData, setPrazoData] = useState<string>(() => {
    56→    const d = new Date();
    57→    d.setDate(d.getDate() + 7);
    58→    return d.toISOString().split('T')[0];
    59→  });
    60→  const [valorRecebido, setValorRecebido] = useState<string>('');
    61→  const [selectedPixSlot, setSelectedPixSlot] = useState<1 | 2>(1);
    62→  const [isTrocaActive, setIsTrocaActive] = useState(false);
    63→  const [valorTroca, setValorTroca] = useState('');
    64→  const [printWidth, setPrintWidth] = useState<'56MM' | '80MM'>('56MM');
    65→
    66→  useEffect(() => {
    67→    if (activeCategoryId && !visitedCategoryIds.includes(activeCategoryId)) {
    68→      setVisitedCategoryIds(prev => [...prev, activeCategoryId]);
    69→    }
    70→  }, [activeCategoryId, visitedCategoryIds]);
    71→
    72→  useEffect(() => {
    73→    if (activeCategoryId === '' && categories.length > 0) {
    74→      setActiveCategoryId(categories[0].id);
    75→    }
    76→  }, [categories, activeCategoryId]);
    77→
    78→  useEffect(() => {
    79→    saveLocalState(cartKey, cart);
    80→  }, [cart, cartKey]);
    81→
    82→  const allCategoriesVisited = visitedCategoryIds.length >= categories.length;

   100→    if (n.includes('SALTY')) return 'SALTY';

   109→      .map(p => ({
   110→        produtoId: p.id,
   111→        quantidade: cart[p.id].quantidade,

   120→

   122→  // FUNÇÃO DE VALIDAÇÃO DIRETA - CHAMA EM QUALQUER LUGAR
   123→  // ============================================================
   124→  const checkPrecoMinimoViolation = (): { hasViolation: boolean; violatingProducts: Array<{ nome: string; preco: number; minimo: number }> } => {
   125→    if (!margemMinimaAtiva || isPrePedido) return { hasViolation: false, violatingProducts: [] };

   130→      if (item.quantidade <= 0) return;
   131→      const p = products.find(prod => prod.id === pId);
   132→      if (!p) return;
   133→      
   134→      const price = parseFloat(item.precoVenda) || 0;
   135→      const minPrice = p.precoMinimo || 0;
   136→      
   137→      // Só valida se tem preço mínimo CADASTRADO (> 0)
   138→      if (minPrice > 0 && price < minPrice) {
   139→        violating.push({ 
   140→          nome: p.nome, 
   141→          preco: price, 
   142→          minimo: minPrice 
   143→        });
   144→      }
   145→    });
   146→    
   147→    return { hasViolation: violating.length > 0, violatingProducts: violating };
   148→  };
   149→
   150→  // Para compatibilidade com código existente
   151→  const hasMarginViolation = checkPrecoMinimoViolation().hasViolation;
   152→
   153→  const updateCart = (pId: string, delta: number, basePrice: number) => {
   154→    setCart(prev => {
   155→      const current = prev[pId] || { quantidade: 0, precoVenda: basePrice.toString() };
   156→      const maxLimit = isPrePedido 
   157→        ? (products.find(p => p.id === pId)?.estoquePrincipal || 0)
   158→        : (minhaCarga.find(c => c.produtoId === pId)?.quantidade || 0);
   159→      const novaQtd = Math.max(0, Math.min(maxLimit, (current.quantidade ?? 0) + delta)); 
   160→      if (novaQtd === 0 && delta < 0) {
   161→        const { [pId]: _, ...rest } = prev;
   162→        return rest;
   163→      }
   164→      return { ...prev, [pId]: { ...current, quantidade: novaQtd } };
   165→    });
   166→  };
   167→
   168→  // Validação no blur - corrige automaticamente se abaixo do mínimo
   169→  const handlePriceBlur = (pId: string, value: string) => {
   170→    if (!margemMinimaAtiva || isPrePedido) return;
   171→    
   172→    const sanitized = value.replace(',', '.');
   173→    const p = products.find(prod => prod.id === pId);
   174→    const minPrice = p?.precoMinimo || 0;
   175→    
   176→    if (sanitized && minPrice > 0) {
   177→      const numValue = parseFloat(sanitized);
   178→      if (!isNaN(numValue) && numValue < minPrice) {
   179→        setCart(prev => {
   180→          if (!prev[pId]) return prev;
   181→          return { ...prev, [pId]: { ...prev[pId], precoVenda: minPrice.toFixed(2) } };
   182→        });
   183→      }
   184→    }
   185→  };
   186→
   187→  // Mudança no input - apenas atualiza, não bloqueia digitação
   188→  const handlePriceChange = (pId: string, value: string) => {
   189→    const sanitized = value.replace(',', '.');
   190→    setCart(prev => {
   191→      if (!prev[pId]) return prev;
   192→      return { ...prev, [pId]: { ...prev[pId], precoVenda: sanitized } };
   193→    });
   194→  };
   195→
   196→  const subtotal = useMemo(() => {
   197→    return Object.entries(cart).reduce((acc, [_, item]) => {
   198→      const cartItem = item as CartItem;
   199→      const price = parseFloat(cartItem.precoVenda) || 0;
   200→      return acc + ((cartItem.quantidade ?? 0) * price); 
   201→    }, 0);
   202→  }, [cart]);
   203→
   204→  const total = useMemo(() => {
   205→    const vt = parseFloat(valorTroca) || 0;
   206→    return Math.max(0, subtotal - (isTrocaActive ? vt : 0));
   207→  }, [subtotal, isTrocaActive, valorTroca]);
   208→
   209→  const troco = useMemo(() => {
   210→    const rec = parseFloat(valorRecebido) || 0;
   211→    return Math.max(0, rec - total);
   212→  }, [valorRecebido, total]);
   213→
   214→  const setPrazoDays = (days: number) => {
   215→    const d = new Date();
   216→    d.setDate(d.getDate() + days);
   217→    setPrazoData(d.toISOString().split('T')[0]);
   218→  };
   219→
   220→  // ============================================================
   221→  // BLOQUEIO INQUEBRÁVEL - CHAMADO ANTES DE QUALQUER FINALIZAÇÃO
   222→  // ============================================================
   223→  const validateAndBlockIfNeeded = (): boolean => {
   224→    const { hasViolation, violatingProducts } = checkPrecoMinimoViolation();
   225→    
   226→    if (hasViolation) {
   227→      const detalhes = violatingProducts
   228→        .map(v => `• ${v.nome}: digitou R$ ${v.preco.toFixed(2)}, mínimo é R$ ${v.minimo.toFixed(2)}`)
   229→        .join('\n');
   230→      
   231→      alert(
   232→        `❌ VENDA BLOQUEADA - PREÇO ABAIXO DO MÍNIMO\n\n` +
   233→        `${detalhes}\n\n` +
   234→        `Corrija os preços antes de continuar.`
   235→      );
   236→      return false; // BLOQUEIA
   237→    }
   238→    return true; // LIBERA
   239→  };
   240→
   241→  const handleConfirmFinalize = async () => {
   242→    if (total <= 0 && getOrderedItems().length === 0) return;
   243→    
   244→    // TRAVA ABSOLUTA - não passa daqui se houver violação
   245→    if (!validateAndBlockIfNeeded()) return;
   246→
   247→    const itens = getOrderedItems();
   248→    const vt = parseFloat(valorTroca) || 0;
   249→    const descontoInfo = isTrocaActive && vt > 0 ? ` (Troca: R$ ${vt.toFixed(2)})` : '';
   250→    
   251→    let finalDetalhe = '';
   252→    if (metodo === 'PIX') {
   253→      finalDetalhe = selectedPixSlot === 1 ? pix1Name : pix2Name;
   254→    } else if (metodo === 'A_PRAZO') {
   255→      const tipoLabel = tipoPrazo === 'PRAZO_COMUM' ? 'Prazo Comum' : tipoPrazo === 'CHEQUE' ? 'Cheque' : 'Boleto';
   256→      finalDetalhe = tipoLabel;
   257→    } else {
   258→      finalDetalhe = detalheMetodo;
   259→    }
   260→    finalDetalhe += descontoInfo;
   261→    
   262→    const salePayload: any = { 
   263→      vendedorId, 
   264→      clientId: client.id, 
   265→      valorTotal: total, 
   266→      valorPago: metodo === 'A_PRAZO' ? 0 : total,
   267→      metodoPagamento: metodo, 
   268→      detalhePagamento: finalDetalhe, 
   269→      itens,
   270→      data: new Date() 
   271→    };
   272→
   273→    if (metodo === 'A_PRAZO') { 
   274→      salePayload.statusPagamento = 'PENDENTE'; 
   275→      salePayload.dataVencimento = new Date(prazoData); 
   276→    } else { 
   277→      salePayload.statusPagamento = 'PAGO'; 
   278→    }
   279→    
   280→    const newSale = await processSale(salePayload);
   281→    if (newSale) { 
   282→      localStorage.removeItem(cartKey); 
   283→      onFinish(newSale); 
   284→    }
   285→  };
   286→
   287→  const productIdsInCarga = useMemo(() => new Set(minhaCarga.map(c => c.produtoId)), [minhaCarga]);
   288→  
   289→  const filteredProducts = useMemo(() => {
   290→    if (isPrePedido) {
   291→      const activeProds = products.filter(p => p.ativo);
   292→      if (!activeCategoryId) return activeProds;
   293→      return activeProds.filter(p => p.categoryId === activeCategoryId);
   294→    }
   295→    const productsInCarga = products.filter(p => productIdsInCarga.has(p.id));
   296→    if (!activeCategoryId) return productsInCarga;
   297→    return productsInCarga.filter(p => p.categoryId === activeCategoryId);
   298→  }, [products, productIdsInCarga, activeCategoryId, isPrePedido]);
   299→
   300→  const generatePrePedidoText = (width: '56MM' | '80MM') => {
   301→    const totalWidth = width === '80MM' ? 48 : 32; 
   302→    
   303→    const padR = (str: string, len: number) => str.substring(0, len).padEnd(len);
   304→    const padL = (str: string, len: number) => str.substring(0, len).padStart(len);
   305→    const center = (str: string, len: number) => {
   306→      const s = str.substring(0, len);
   307→      const spaces = Math.max(0, Math.floor((len - s.length) / 2));
   308→      return ' '.repeat(spaces) + s;
   309→    };
   310→
   311→    let t = '';
   312→    
   313→    t += '*'.repeat(totalWidth) + '\n';
   314→    t += center('PRÉ-PEDIDO (RASCUNHO)', totalWidth) + '\n';
   315→    t += '*'.repeat(totalWidth) + '\n';
   316→    
   317→    const clientName = client.nomeFantasia || 'Consumidor';
   318→    t += `Cliente: ${clientName}\n`;
   319→    t += `Data: ${new Date().toLocaleDateString()} ${new Date().toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'})}\n`;
   320→    t += '-'.repeat(totalWidth) + '\n';
   321→
   322→    const qtyW = 4;
   323→    const valW = width === '80MM' ? 13 : 8;
   324→    const descW = totalWidth - qtyW - valW;
   325→
   326→    t += padR('DESCRICAO', descW) + padL('QTD', qtyW) + padL('VALOR', valW) + '\n';
   327→    t += '-'.repeat(totalWidth) + '\n';
   328→
   329→    const items = getOrderedItems();
   330→    items.forEach(item => {
   331→      const p = products.find(prod => prod.id === item.produtoId);
   332→      const productName = (p?.nome ?? 'Produto');
   333→      
   334→      const qtyStr = `${item.quantidade}x`;
   335→      const valStr = `${(item.quantidade * item.precoVenda).toFixed(2)}`;
   336→
   337→      t += padR(productName.substring(0, descW), descW) + padL(qtyStr, qtyW) + padL(valStr, valW) + '\n';
   338→
   339→      let remaining = productName.substring(descW);
   340→      while (remaining.length > 0) {
   341→        t += padR(remaining.substring(0, totalWidth), totalWidth) + '\n';
   342→        remaining = remaining.substring(totalWidth);
   343→      }
   344→    });
   345→
   346→    t += '-'.repeat(totalWidth) + '\n';
   347→    
   348→    const totalLabel = 'TOTAL ESTIMADO:';
   349→    const totalVal = `R$ ${total.toFixed(2)}`;
   350→    t += padR(totalLabel, totalWidth - totalVal.length) + totalVal + '\n';
   351→
   352→    t += '-'.repeat(totalWidth) + '\n';
   353→    t += center('ATENÇÃO: ESTE DOCUMENTO', totalWidth) + '\n';
   354→    t += center('NÃO É UMA VENDA REAL.', totalWidth) + '\n';
   355→    t += center('APENAS RESERVA/PLANEJAMENTO', totalWidth) + '\n';
   356→    t += '*'.repeat(totalWidth) + '\n';
   357→    t += '\n\n\n\n\n';
   358→
   359→    return t;
   360→  };
   361→
   362→  const handlePrintPrePedido = async () => {
   363→    const rawText = generatePrePedidoText(printWidth);

   365→      await printerService.printNative(rawText);

   379→      <div className="fixed inset-0 bg-gray-900 z-[100] flex flex-col">

   381→          <Cupom

   383→              id: 'preview', 
   384→              vendedorId, 
   385→              clientId: client.id, 
   386→              data: new Date(), 
   387→              valorTotal: total, 
   388→              valorPago: 0, 
   389→              metodoPagamento: metodo, 
   390→              statusPagamento: metodo === 'A_PRAZO' ? 'PENDENTE' : 'PAGO',

   392→            }} 
   393→            client={client} 
   394→            products={products} 
   395→            onBack={() => setView('CART')} 
   396→            onClose={() => setView('PAYMENT')}

   403→  }

   408→        <div className="bg-white w-full max-w-[340px] shadow-2xl flex flex-col animate-in zoom-in-95 duration-300 relative rounded-t-3xl overflow-hidden">
   409→          
   410→          <div className="p-6 bg-white overflow-hidden mt-4">
   411→            <h3 className="text-center font-black text-xs text-indigo-600 uppercase mb-4 tracking-widest"><i className="fa-solid fa-file-invoice mr-2"></i> Pré-Pedido Rascunho</h3>
   412→            <div className="font-mono text-[11px] leading-tight text-black bg-white whitespace-pre select-none border-l-2 border-indigo-100 pl-4">
   413→              {generatePrePedidoText(printWidth)}
   414→            </div>
   415→          </div>
   416→
   417→          <div className="bg-gray-100 p-5 flex flex-col gap-3 border-t border-gray-200">
   418→            <div className="flex bg-gray-200 p-1 rounded-2xl mb-1">
   419→              <button onClick={() => setPrintWidth('56MM')} className={`flex-1 py-2 rounded-xl text-[10px] font-black uppercase transition-all ${printWidth === '56MM' ? 'bg-white shadow-sm text-indigo-600' : 'text-gray-500'}`}>56mm</button>
   420→              <button onClick={() => setPrintWidth('80MM')} className={`flex-1 py-2 rounded-xl text-[10px] font-black uppercase transition-all ${printWidth === '80MM' ? 'bg-white shadow-sm text-indigo-600' : 'text-gray-400'}`}>80mm</button>
   421→            </div>
   422→            
   423→            <div className="grid grid-cols-2 gap-2">
   424→              <button onClick={handlePrintPrePedido} className="bg-indigo-600 text-white font-black py-4 rounded-2xl flex items-center justify-center gap-2 active:scale-95 text-[10px] uppercase shadow-lg">
   425→                <i className="fa-solid fa-print"></i> Imprimir
   426→              </button>
   427→              <button onClick={handleCopyPrePedido} className="bg-emerald-600 text-white font-black py-4 rounded-2xl flex items-center justify-center gap-2 active:scale-95 text-[10px] uppercase shadow-lg">
   428→                <i className="fa-solid fa-copy"></i> Copiar

   431→
   432→            <button

   435→                setView('CART');

   437→              }} 
   438→              className="w-full bg-slate-900 text-white font-black py-4 rounded-2xl active:scale-95 text-[10px] uppercase tracking-widest shadow-xl"
   439→            >
   440→              VOLTAR AO ATENDIMENTO

   442→          </div>
   443→        </div>

   447→
   448→  return (
   449→    <div className="fixed inset-0 bg-gray-50 z-[60] flex flex-col">

   453→          <div className="text-center px-4 truncate">
   454→            <p className="text-[9px] text-gray-400 uppercase font-black tracking-tighter">
   455→              {isPrePedido ? 'Pré-Pedido Rascunho' : 'Atendimento'}

   459→          <button onClick={() => setCart({})} className="w-9 h-9 bg-rose-50 text-rose-400 rounded-xl flex items-center justify-center active:scale-90 transition-transform"><i className="fa-solid fa-trash-can text-sm"></i></button>
   460→        </div>
   461→
   462→        <div className="grid grid-cols-4 gap-1 px-0.5">
   463→          {categories.map(cat => {
   464→            const isVisited = visitedCategoryIds.includes(cat.id);
   465→            return (
   466→              <button 
   467→                key={cat.id}
   468→                onClick={() => setActiveCategoryId(cat.id)}
   469→                className={`py-1.5 px-1 rounded-lg text-[8px] font-black uppercase transition-all truncate border relative ${activeCategoryId === cat.id ? 'bg-blue-600 text-white border-blue-600 shadow-sm' : 'bg-gray-50 text-gray-400 border-gray-100'}`}
   470→              >

   570→

   603→        <div className="flex justify-between items-end px-1">

   606→              {isPrePedido ? 'Total Rascunho' : 'Total Líquido'}

   619→            className={`px-6 py-4 rounded-2xl font-black uppercase text-xs ${(total > 0 || getOrderedItems().length > 0) && !hasMarginViolation && (allCategoriesVisited || isPrePedido) ? (isPrePedido ? 'bg-indigo-600 text-white shadow-lg' : 'bg-blue-600 text-white shadow-lg') : 'bg-gray-200 text-gray-400'}`}

   628→           <div className="bg-white w-full max-sm rounded-[2.5rem] p-8 animate-in slide-in-from-bottom max-h-[95vh] overflow-y-auto">

   697→                   </div>

   699→                   <div className="bg-gray-50 p-4 rounded-2xl flex flex-col items-center border border-gray-100">
   700→                     <p className="text-[8px] font-black text-gray-400 uppercase mb-3">QR CODE PARA PAGAMENTO</p>

   702→                        {selectedPixSlot === 1 ? (

   716→              >
   717→                Concluir Venda

   719→              <button onClick={() => setView('RECEIPT_PREVIEW')} className="w-full py-3 text-gray-400 font-bold text-[9px] uppercase tracking-widest">Voltar ao Cupom</button>
   720→           </div>
   721→        </div>
   722→      )}
   723→    </div>
   724→  );
   725→};
