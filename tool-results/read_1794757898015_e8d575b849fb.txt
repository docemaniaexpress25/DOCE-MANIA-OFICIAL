    87→/** Rótulo certo do documento: NFCE -> "NFC-e", NFE -> "NF-e" (regra BLOCO 135). */
    88→function rotuloTipo(tipo: string, modoNota: string): string {
    89→  const t = String(tipo || '').toUpperCase();
    90→  if (t === 'NFCE') return 'NFC-e';
    91→  if (t === 'NFE') return 'NF-e';
    92→  const m = String(modoNota || '').toUpperCase();
    93→  if (m === 'NFE') return 'NF-e';
    94→  if (m === 'NFCE') return 'NFC-e';
    95→  return 'NF';
    96→}

   101→const PERIODO_META: { key: FiltroPeriodo; label: string }[] = [
   102→  { key: 'HOJE', label: 'Hoje' },
   103→  { key: '7D', label: '7 dias' },
   104→  { key: '30D', label: '30 dias' },

   108→
   109→function dentroPeriodo(dataVenda: string, p: FiltroPeriodo): boolean {
   110→  if (p === 'TUDO' || !dataVenda) return true;
   111→  const d = new Date(dataVenda);
   112→  if (!isFinite(d.getTime())) return true;
   113→  const agora = new Date();
   114→  const tz = 'America/Sao_Paulo';
   115→  const diaStr = (x: Date) => new Intl.DateTimeFormat('en-CA', { timeZone: tz, year: 'numeric', month: '2-digit', day: '2-digit' }).format(x);
   116→  if (p === 'HOJE') return diaStr(d) === diaStr(agora);
   117→  if (p === 'MES') return diaStr(d).slice(0, 7) === diaStr(agora).slice(0, 7);
   118→  const dias = p === '7D' ? 7 : 30;
   119→  return d.getTime() >= agora.getTime() - dias * 24 * 60 * 60 * 1000;
   120→}
