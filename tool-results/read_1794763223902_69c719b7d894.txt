   132→/** Rótulo certo do documento: NFCE -> "NFC-e", NFE -> "NF-e" (regra BLOCO 135). */

   134→  const t = String(tipo || '').toUpperCase();
   135→  if (t === 'NFCE') return 'NFC-e';

   154→function dentroPeriodo(dataVenda: string, p: FiltroPeriodo): boolean {
   155→  if (p === 'TUDO' || !dataVenda) return true;
   156→  const d = new Date(dataVenda);
   157→  if (!isFinite(d.getTime())) return true;
   158→  const agora = new Date();
   159→  const tz = 'America/Sao_Paulo';
   160→  const diaStr = (x: Date) => new Intl.DateTimeFormat('en-CA', { timeZone: tz, year: 'numeric', month: '2-digit', day: '2-digit' }).format(x);
   161→  if (p === 'HOJE') return diaStr(d) === diaStr(agora);
   162→  if (p === 'MES') return diaStr(d).slice(0, 7) === diaStr(agora).slice(0, 7);
   163→  const dias = p === '7D' ? 7 : 30;
   164→  return d.getTime() >= agora.getTime() - dias * 24 * 60 * 60 * 1000;
   165→}

   188→/** ArrayBuffer -> base64 (chunks para não estourar a pilha com XMLs grandes). */
   189→function bytesToBase64(u8: Uint8Array): string {
   190→  let bin = '';
   191→  const CH = 0x8000;
   192→  for (let i = 0; i < u8.length; i += CH) {
   193→    bin += String.fromCharCode(...Array.from(u8.subarray(i, i + CH)));
   194→  }
   195→  return btoa(bin);
   196→}

   669→    const add: { nome: string; base64: string; tamanho: number }[] = [];
   670→    const zipTravados: string[] = [];
   671→    for (const f of Array.from(files).slice(0, 200)) {
   672→      try {
   673→        if (/\.zip$/i.test(f.name)) {
   674→          const zipped = unzipSync(new Uint8Array(await f.arrayBuffer()));
   675→          for (const [nome, u8] of Object.entries(zipped)) {
   676→            if (!/\.xml$/i.test(nome) || nome.startsWith('__MACOSX')) continue;
   677→            add.push({ nome: `${f.name}/${nome}`, base64: bytesToBase64(u8), tamanho: u8.length });
   678→          }
   679→        } else if (/\.xml$/i.test(f.name) || f.type === 'text/xml' || f.type === 'application/xml') {
   680→          const u8 = new Uint8Array(await f.arrayBuffer());
   681→          add.push({ nome: f.name, base64: bytesToBase64(u8), tamanho: u8.length });
   682→        } else {
   683→          zipTravados.push(f.name);
   684→        }
   685→      } catch {
   686→        zipTravados.push(f.name);
   687→      }
   688→    }
   689→    setXmlArquivos(atuais => {
   690→      const nomes = new Set(atuais.map(a => a.nome));
   691→      const novos = add.filter(a => !nomes.has(a.nome) && a.tamanho > 0);
   692→      return [...atuais, ...novos];
   693→    });
   694→    if (zipTravados.length) showToast(`${zipTravados.length} arquivo(s) ignorado(s) — só aceito .xml e .zip.`, false);
   695→    if (xmlRes) setXmlRes(null);
