   138→  const checkPrecoMinimoViolation = (): { hasViolation: boolean; violatingProducts: Array<{ nome: string; preco: number; minimo: number }> } => {

   145→      const p = products.find(prod => prod.id === pId);
   146→      if (!p) return;
   147→      
   148→      const price = parseFloat(item.precoVenda) || 0;
   149→      const minPrice = p.precoMinimo || 0;
   150→      
   151→      // Só valida se tem preço mínimo CADASTRADO (> 0)
   152→      if (minPrice > 0 && price < minPrice) {
   153→        violating.push({ 
   154→          nome: p.nome, 
   155→          preco: price, 
   156→          minimo: minPrice 
   157→        });
   158→      }
   159→    });
   160→    
   161→    return { hasViolation: violating.length > 0, violatingProducts: violating };
   162→  };

   169→      const current = prev[pId] || { quantidade: 0, precoVenda: basePrice.toString() };

   174→      if (novaQtd === 0 && delta < 0) {
   175→        const { [pId]: _, ...rest } = prev;
   176→        return rest;
   177→      }
   178→      return { ...prev, [pId]: { ...current, quantidade: novaQtd } };
   179→    });
   180→  };
   181→

   184→    if (!margemMinimaAtiva || isPrePedido) return;
   185→    
   186→    const sanitized = value.replace(',', '.');
   187→    const p = products.find(prod => prod.id === pId);
   188→    const minPrice = p?.precoMinimo || 0;
   189→    
   190→    if (sanitized && minPrice > 0) {
   191→      const numValue = parseFloat(sanitized);
   192→      if (!isNaN(numValue) && numValue < minPrice) {
   193→        setCart(prev => {
   194→          if (!prev[pId]) return prev;
   195→          return { ...prev, [pId]: { ...prev[pId], precoVenda: minPrice.toFixed(2) } };
   196→        });
   197→      }
   198→    }
   199→  };
   200→
   201→  // Mudança no input - apenas atualiza, não bloqueia digitação
   202→  const handlePriceChange = (pId: string, value: string) => {
   203→    const sanitized = value.replace(',', '.');
   204→    setCart(prev => {
   205→      if (!prev[pId]) return prev;
   206→      return { ...prev, [pId]: { ...prev[pId], precoVenda: sanitized } };
   207→    });

   211→    return Object.entries(cart).reduce((acc, [_, item]) => {
   212→      const cartItem = item as CartItem;
   213→      const price = parseFloat(cartItem.precoVenda) || 0;

   216→  }, [cart]);

   219→    const vt = parseFloat(valorTroca) || 0;

   225→    return Math.max(0, rec - total);
   226→  }, [valorRecebido, total]);
   227→
   228→  const setPrazoDays = (days: number) => {
   229→    const d = new Date();
   230→    d.setDate(d.getDate() + days);

   238→    const { hasViolation, violatingProducts } = checkPrecoMinimoViolation();

   240→    if (hasViolation) {
   241→      const detalhes = violatingProducts
   242→        .map(v => `• ${v.nome}: digitou R$ ${v.preco.toFixed(2)}, mínimo é R$ ${v.minimo.toFixed(2)}`)

   246→        `❌ VENDA BLOQUEADA - PREÇO ABAIXO DO MÍNIMO\n\n` +

   248→        `Corrija os preços antes de continuar.`

   259→    if (!validateAndBlockIfNeeded()) return;
   260→
   261→    const itens = getOrderedItems();
   262→    const vt = parseFloat(valorTroca) || 0;
   263→    const descontoInfo = isTrocaActive && vt > 0 ? ` (Troca: R$ ${vt.toFixed(2)})` : '';

   265→    let finalDetalhe = '';
   266→    if (metodo === 'PIX') {

   269→      const tipoLabel = tipoPrazo === 'PRAZO_COMUM' ? 'Prazo Comum' : tipoPrazo === 'CHEQUE' ? 'Cheque' : 'Boleto';

   272→      finalDetalhe = detalheMetodo;

   331→  const generatePrePedidoText = (width: '56MM' | '80MM') => {
   332→    const totalWidth = width === '80MM' ? 48 : 32; 
   333→    
   334→    const padR = (str: string, len: number) => str.substring(0, len).padEnd(len);
   335→    const padL = (str: string, len: number) => str.substring(0, len).padStart(len);
   336→    const center = (str: string, len: number) => {
   337→      const s = str.substring(0, len);
   338→      const spaces = Math.max(0, Math.floor((len - s.length) / 2));
   339→      return ' '.repeat(spaces) + s;
   340→    };
   341→
   342→    let t = '';
   343→    
   344→    t += '*'.repeat(totalWidth) + '\n';

   350→    t += `Data: ${new Date().toLocaleDateString()} ${new Date().toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'})}\n`;
   351→    t += '-'.repeat(totalWidth) + '\n';
   352→
   353→    const qtyW = 4;
   354→    const valW = width === '80MM' ? 13 : 8;
   355→    const descW = totalWidth - qtyW - valW;
   356→
   357→    t += padR('DESCRICAO', descW) + padL('QTD', qtyW) + padL('VALOR', valW) + '\n';

   366→      const valStr = `${(item.quantidade * item.precoVenda).toFixed(2)}`;

   368→      t += padR(productName.substring(0, descW), descW) + padL(qtyStr, qtyW) + padL(valStr, valW) + '\n';
   369→
   370→      let remaining = productName.substring(descW);
   371→      while (remaining.length > 0) {
   372→        t += padR(remaining.substring(0, totalWidth), totalWidth) + '\n';
   373→        remaining = remaining.substring(totalWidth);
   374→      }
   375→    });
   376→
   377→    t += '-'.repeat(totalWidth) + '\n';

   379→    const totalLabel = 'TOTAL ESTIMADO:';

   384→    t += center('ATENÇÃO: ESTE DOCUMENTO', totalWidth) + '\n';

   388→    t += '\n\n\n\n\n';

   453→
   454→  if (view === 'PRE_PEDIDO_PREVIEW') {

   457→        <div className="bg-white w-full max-w-[340px] shadow-2xl flex flex-col animate-in zoom-in-95 duration-300 relative rounded-t-3xl overflow-hidden">
   458→          
   459→          <div className="p-6 bg-white overflow-hidden mt-4">
   460→            <h3 className="text-center font-black text-xs text-indigo-600 uppercase mb-4 tracking-widest"><i className="fa-solid fa-file-invoice mr-2"></i> Pré-Pedido Rascunho</h3>
