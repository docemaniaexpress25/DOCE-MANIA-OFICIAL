    12→ * - Pedido mínimo de atacado (MIN_PEDIDO) — igual ao site antigo (R$ 200);
    13→ * - Checkout com RECIBO estilo cupom + forma de pagamento (Pix/Dinheiro);
    14→ * - AO ENVIAR: o pedido nasce como PRÉ-VENDA na FILA do sistema (separação),
    15→ *   com preço/estoque SEMPRE do banco — NÃO vai para WhatsApp como o site antigo.
    16→ *
    17→ * O código na URL é o portal_code aleatório do cliente (mesmo link do portal).
    18→ */

    29→
    30→interface Categoria { id: string; name: string; }
    31→interface ClienteInfo { nomeCompleto: string; bairro: string; endereco: string; }
    32→
    33→/** Pedido mínimo de atacado (igual ao site antigo). Fácil de mexer. */
    34→const MIN_PEDIDO = 200;
    35→/** Cor do site antigo (verde Doce Mania) */
    36→const PRIMARY = "#13ec5b";
    37→const PRIMARY_DARK = "#0fb847";
    38→
    39→const fmt = (v: number) => v.toLocaleString("pt-BR", { style: "currency", currency: "BRL" });
    40→const semAcento = (s: string) => s.normalize("NFD").replace(/[\u0300-\u036f]/g, "");
    41→
    42→/** Foto da família: cadastrada (Bloco 15) > catálogo antigo > null (placeholder) */
    43→const fotoDe = (f: FamiliaCatalogo, produtoImagem: Map<string, string | null>): string | null => {
    44→  for (const v of f.variantes) {
    45→    const img = produtoImagem.get(v.produtoId);
    46→    if (img) return img;
    47→  }
    48→  return fotoFamilia(f.nome);
    49→};

    60→  const [bloqueado, setBloqueado] = useState(false);

    62→
    63→  const [carrinho, setCarrinho] = useState<Record<string, number>>({});

    90→      setErro("Sem conexao. Verifique a internet e tente de novo.");

   101→    try {
   102→      const saved = localStorage.getItem(CART_KEY);
   103→      if (saved) {
   104→        const parsed = JSON.parse(saved) as Record<string, number>;
   105→        if (parsed && typeof parsed === "object") setCarrinho(parsed);
   106→      }

   108→  }, []);
   109→  useEffect(() => {
   110→    try { localStorage.setItem(CART_KEY, JSON.stringify(carrinho)); } catch { /* ignora */ }
   111→  }, [carrinho, CART_KEY]);
   112→

   138→    });

   141→  /** Quantidade (em UN) de toda a familia no carrinho */
   142→  const qtdFamilia = useCallback((f: FamiliaCatalogo) =>

   146→  const addQty = (p: ProdutoCatalogo, upb: number) => {

   148→    setCarrinho(prev => {
   149→      const atual = prev[p.id] || 0;

   151→      const q = Math.min(nova, Math.max(0, Math.min(p.estoque, atual + passo)));
   152→      const copy = { ...prev };
   153→      if (q <= 0) delete copy[p.id]; else copy[p.id] = q;
   154→      return copy;
   155→    });
   156→  };

   160→      const atual = prev[p.id] || 0;
   161→      const q = Math.max(0, atual - passo);

   163→      if (q <= 0) delete copy[p.id]; else copy[p.id] = q;
   164→      return copy;
   165→    });

   172→      (entries) => setCheckoutVisivel(!!entries[0]?.isIntersecting),

   222→

   226→      <div className="min-h-screen bg-white flex flex-col items-center justify-center p-6 text-center">

   231→  }
   232→
   233→  if (erro || !client) {
   234→    return (
   235→      <div className="min-h-screen bg-white flex flex-col items-center justify-center px-6 text-center">
   236→        <h1 className="font-black text-gray-900 text-lg mb-1">Ops!</h1>
   237→        <p className="text-xs text-neutral-500 font-semibold mb-6">{erro || "Link invalido."}</p>
   238→        <button onClick={carregar} className="bg-[#13ec5b] text-black font-black text-[10px] uppercase tracking-widest px-6 py-3.5 rounded-full shadow-lg shadow-[#13ec5b]/30 active:scale-95 transition-transform">
   239→          Tentar de novo
   240→        </button>

   253→        <a href={`/cliente/${encodeURIComponent(codigo)}`} className="bg-[#13ec5b] text-black font-black text-[10px] uppercase tracking-widest px-6 py-3.5 rounded-full shadow-lg shadow-[#13ec5b]/30 active:scale-95 transition-transform">
   254→          Ir para o portal

   264→      {/* ===== HEADER (igual site antigo) ===== */}

   266→        <div className="flex items-center gap-3 min-w-0">
   267→          <div className="w-11 h-11 shrink-0 bg-[#13ec5b] rounded-2xl flex items-center justify-center shadow-sm">
   268→            {/* A logo.png do site antigo nao esta mais acessivel — marca em texto */}

   270→          </div>
   271→          <div className="flex flex-col min-w-0">

   285→          <h2 className="text-[10px] font-black uppercase tracking-widest text-[#0fb847]">
   286→            🔥 Fornecedor de Salgadinhos Elma Chips em Caxias do Sul 🔥

   300→          </div>
