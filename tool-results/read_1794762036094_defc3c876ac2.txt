   117→   117→

   119→   119→  if (!saleId) return BAD(400, 'Parametro ausente (saleId).');
   120→   120→

   143→   143→export async function POST(req: NextRequest) {
   144→   144→  const bridged = await devBridge(req);
   145→   145→  if (bridged) return bridged;
   146→   146→  const session = sessionFromRequest(req);
   147→   147→  if (!session) return BAD(401, 'Sessao expirada. Faca login novamente.');
   148→   148→
   149→   149→  const body = await req.json().catch(() => ({}));
   150→   150→  const acao = body.acao as string;
   151→   151→  const saleId = body.saleId as string;
   152→   152→  if (!saleId || !acao) return BAD(400, 'Parametros ausentes (acao, saleId).');
   153→   153→
   154→   154→  if (!(await hasNotaColumns())) {
   155→   155→    return BAD(503, 'Colunas de nota fiscal nao existem ainda no Supabase. Rode o SQL do Bloco 10 (docs/NOTAS-FISCAIS.md).');
   156→   156→  }
   157→   157→
   158→   158→  const db = getServiceClient();
   159→   159→
   160→   160→  // Carrega a venda e valida permissao
   161→   161→  const { data: sale, error: saleErr } = await db
   162→   162→    .from('sales')
   163→   163→    .select(await colsNota())
   164→   164→    .eq('id', saleId)
   165→   165→    .single();
   166→   166→
   167→   167→  if (saleErr || !sale) return BAD(404, 'Venda nao encontrada.');
   168→   168→  if (session.perfil !== 'ADMIN' && session.sub !== sale.vendedor_id) {
   169→   169→    return BAD(403, 'Voce so pode emitir nota para suas proprias vendas.');
   170→   170→  }
   171→   171→
   172→   172→  // ---------- CONSULTAR ----------
   173→   173→  if (acao === 'CONSULTAR') {
   174→   174→    if (!sale.nota_ref) return BAD(422, 'Esta venda ainda nao tem nota emitida.');
   175→   175→    if (!isFocusConfigured()) return BAD(503, 'Focus NFe nao configurada (env FOCUS_NFE_TOKEN no Vercel).');
   176→   176→
   177→   177→    const doc: FocusDoc = sale.nota_tipo === 'NFCE' ? 'nfce' : 'nfe';
   178→   178→    const r = await consultarNota(doc, sale.nota_ref as string);
   179→   179→    if (!r.ok || !r.data || 'erro' in r.data) {
   180→   180→      return BAD(502, (r.data as { erro?: string }).erro || 'Falha ao consultar a Focus NFe.', { nota: notaPublica(sale as Record<string, unknown>) });
   181→   181→    }
   182→   182→    const res = interpretarResultado(r.data as Record<string, unknown>);
   183→   183→    const patch: Record<string, unknown> = {
   184→   184→      nota_status: res.autorizada ? 'AUTORIZADA' : res.cancelada ? 'CANCELADA' : res.motivo ? 'REJEITADA' : 'EMITINDO',
   185→   185→      nota_numero: res.numero || sale.nota_numero || null,
   186→   186→      nota_pdf_url: res.pdfUrl || sale.nota_pdf_url || null,
   187→   187→      nota_xml_url: res.xmlUrl || sale.nota_xml_url || null,
   188→   188→      nota_erro: res.autorizada ? null : res.motivo || null,
   189→   189→    };
   190→   190→    // Bloco 25: guarda a chave de acesso (44 dig) p/ atrelar a NF-e ao boleto Inter
   191→   191→    if (res.chave && (await hasBoletoColumns())) patch.nota_chave = res.chave;
   192→   192→    // BLOCO 142: primeira vez que a SEFAZ confirma a autorização, grava a
   193→   193→    // data-hora oficial — é a BASE do prazo de cancelamento (NFC-e 30min/NF-e 24h)
   194→   194→    if (res.autorizada && res.dataAutorizacao && (await hasNotaEmitidaEm()) && !(sale as Record<string, unknown>).nota_emitida_em) {
   195→   195→      const dAuto = new Date(res.dataAutorizacao);
   196→   196→      if (!isNaN(dAuto.getTime())) patch.nota_emitida_em = dAuto.toISOString();
   197→   197→    }
   198→   198→    await atualizarNota(saleId, patch);
   199→   199→    return NextResponse.json({ ok: true, nota: { ...notaPublica(sale as Record<string, unknown>), ...res } });
   200→   200→  }
   201→   201→
   202→   202→  // ---------- CANCELAR (ADMIN) ----------
   203→   203→  if (acao === 'CANCELAR') {
   204→   204→    if (session.perfil !== 'ADMIN') return BAD(403, 'Somente o admin pode cancelar notas.');
   205→   205→    if (!sale.nota_ref) return BAD(422, 'Esta venda nao tem nota para cancelar.');
   206→   206→    const just = (body.justificativa as string) || '';
   207→   207→    if (just.trim().length < 15) return BAD(422, 'Justificativa obrigatoria (minimo 15 caracteres — regra da SEFAZ).');
   208→   208→    if (!isFocusConfigured()) return BAD(503, 'Focus NFe nao configurada (env FOCUS_NFE_TOKEN no Vercel).');
   209→   209→
   210→   210→    const doc: FocusDoc = sale.nota_tipo === 'NFCE' ? 'nfce' : 'nfe';
   211→   211→
   212→   212→    // ---------------------------------------------------------------
   213→   213→    // BLOCO 142 — GUARDA DO PRAZO SEFAZ (NFC-e 30min / NF-e 24h após a
   214→   214→    // autorização). Passado o prazo a SEFAZ RECUSA o evento — o servidor
   215→   215→    // recusa antes com mensagem clara. Usa a data-hora REAL de autorização:
   216→   216→    // sales.nota_emitida_em; quando ela não existe (venda antiga) consulta
   217→   217→    // a Focus UMA vez (data_autorizacao) e grava. Sem timestamp real o
   218→   218→    // julgamento fica com a SEFAZ — nunca bloqueia por chute (a NF-e pode
   219→   219→    // ter sido batida DIAS depois da venda).
   220→   220→    // ---------------------------------------------------------------
   221→   221→    let prazo: ReturnType<typeof prazoCancelamentoNota> | null = null;
   222→   222→    if (String(sale.nota_status).toUpperCase() === 'AUTORIZADA') {
   223→   223→      let baseAutorizacao = (await hasNotaEmitidaEm()) ? ((sale as Record<string, unknown>).nota_emitida_em as string) || null : null;
   224→   224→      if (!baseAutorizacao && sale.nota_ref) {
   225→   225→        try {
   226→   226→          const c = await consultarNota(doc, sale.nota_ref as string);
   227→   227→          if (c.ok && c.data && !('erro' in c.data)) {
   228→   228→            const ci = interpretarResultado(c.data as Record<string, unknown>);
   229→   229→            if (ci.autorizada && ci.dataAutorizacao) {
   230→   230→              const dAuto = new Date(ci.dataAutorizacao);
   231→   231→              if (!isNaN(dAuto.getTime())) {
   232→   232→                baseAutorizacao = dAuto.toISOString();
   233→   233→                if (await hasNotaEmitidaEm()) await atualizarNota(saleId, { nota_emitida_em: baseAutorizacao });
   234→   234→              }
   235→   235→            }
   236→   236→          }
   237→   237→        } catch { /* Focus indisponível -> a SEFAZ decide no cancelamento */ }
   238→   238→      }
   239→   239→      prazo = prazoCancelamentoNota({ tipo: sale.nota_tipo as string, status: 'AUTORIZADA', emitidaEm: baseAutorizacao });
   240→   240→      if (prazo.expirado && baseAutorizacao) {
   241→   241→        return BAD(
   242→   242→          422,
   243→   243→          `Cancelamento FORA DO PRAZO da SEFAZ — ${doc === 'nfce' ? 'NFC-e' : 'NF-e'} autorizada em ${fmtDataBR(baseAutorizacao)} e o prazo de ${prazo.prazoLabel} já venceu. O evento de cancelamento seria recusado.`,
   244→   244→          { nota: notaPublica(sale as Record<string, unknown>), prazo: { expiraEm: prazo.expiraEm, expirado: true, prazoLabel: prazo.prazoLabel } },
   245→   245→        );
   246→   246→      }
   247→   247→    }
   248→   248→
   249→   249→    const r = await cancelarNota(doc, sale.nota_ref as string, just);
   250→   250→    if (!r.ok || !r.data || 'erro' in r.data) {
   251→   251→      return BAD(502, (r.data as { erro?: string }).erro || 'Falha ao cancelar na Focus NFe.', { nota: notaPublica(sale as Record<string, unknown>) });
   252→   252→    }
   253→   253→    const res = interpretarResultado(r.data as Record<string, unknown>);
   254→   254→    await atualizarNota(saleId, {
   255→   255→      nota_status: res.cancelada ? 'CANCELADA' : sale.nota_status,
   256→   256→      nota_erro: res.cancelada ? null : res.motivo || null,
   257→   257→    });
   258→   258→    // BLOCO 116 (doc reference/cancelar_nfce.md): NFC-e so pode ser
   259→   259→    // cancelada em ate 30 MINUTOS apos a emissao (NF-e: 24h). O aviso vai
   260→   260→    // na resposta para o admin saber por que a SEFAZ pode recusar.
   261→   261→    return NextResponse.json({
   262→   262→      ok: res.cancelada,
   263→   263→      aviso:
   264→   264→        doc === 'nfce'
   265→   265→          ? 'NFC-e so pode ser cancelada em ate 30 minutos apos a emissao (regra da SEFAZ).'
   266→   266→          : undefined,
   267→   267→      prazo: prazo ? { expiraEm: prazo.expiraEm, prazoLabel: prazo.prazoLabel, textoRestante: prazo.textoRestante } : undefined,
   268→   268→      nota: { ...notaPublica(sale as Record<string, unknown>), ...res },
   269→   269→    });
   270→   270→  }
   271→   271→
   272→   272→  // ---------- EMITIR ----------
   273→   273→  if (acao !== 'EMITIR') return BAD(400, `Acao desconhecida: ${acao}`);
   274→   274→
   275→   275→  if (sale.nota_status === 'AUTORIZADA') {
   276→   276→    return NextResponse.json({ ok: true, jaEmitida: true, nota: notaPublica(sale as Record<string, unknown>) });
   277→   277→  }
   278→   278→  if (!isFocusConfigured()) {
   279→   279→    return BAD(503, 'Focus NFe nao configurada. Defina FOCUS_NFE_TOKEN no Vercel (docs/NOTAS-FISCAIS.md).');
   280→   280→  }
   281→   281→
   282→   282→  const cfg = getFocusConfig();
   283→   283→
   284→   284→  // ------------------------------------------------------------------
   285→   285→  // BLOCO 108 — MODO CUPOM (regra do dono): cliente que "quer somente
   286→   286→  // cupom" recebe e paga o cupom normal; a NFC-e ANONIMA que alimenta
   287→   287→  // a fiscal sai com o PREÇO CUPOM do produto (products.preco_cupom;
   288→   288→  // 0 = cai no preço de custo). Liga/desliga nas Configurações ERP.
   289→   289→  // Cliente marcado modo_cupom entra no modo AUTOMATICAMENTE.
   290→   290→  // ------------------------------------------------------------------
   291→   291→  const erp = await lerErpConfig();
   292→   292→  const temPrecoCupom = await hasPrecoCupom();
   293→   293→  const temClientModoCupom = await hasClientModoCupom();
   294→   294→  const temEmitidaEm = await hasNotaEmitidaEm();
   295→   295→  const emitentePayload = (await lerEmitentePayload()) as Partial<FocusNfePayloadV2>;
   296→   296→
   297→   297→  // Cliente da venda
   298→   298→  const { data: client } = await db
   299→   299→    .from('clients')
   300→   300→    .select(`nome_fantasia, nome, cnpj, telefone, endereco, bairro, razao_social, inscricao_estadual, endereco_numero, endereco_cep, endereco_municipio, endereco_uf, email${temClientModoCupom ? ', modo_cupom' : ''}`)
   301→   301→    .eq('id', sale.client_id)
   302→   302→    .single();
   303→   303→
   304→   304→  if (!client) return BAD(422, 'Cliente da venda nao encontrado.');
   305→   305→
   306→   306→  const modoCupomPed = body.modoCupom === true;
   307→   307→  const clientModoCupom = temClientModoCupom && (client as Record<string, unknown>).modo_cupom === true;
   308→   308→  if (modoCupomPed && !erp.cupomDuploNfc) {
   309→   309→    return BAD(422, 'Modo cupom DESLIGADO. Ligue em Configuracoes -> ERP Fiscal & Automacoes (cupomDuploNfc).');
   310→   310→  }
   311→   311→  const modoCupom = erp.cupomDuploNfc && (modoCupomPed || clientModoCupom);
   312→   312→
   313→   313→  // Tipo automatico: CNPJ valido -> NFe; caso contrario NFC-e (consumidor)
   314→   314→  const cpfCnpj = validaCpfCnpj(client.cnpj);
   315→   315→  const tipoEscolhido = (body.tipo as string) || '';
   316→   316→  let doc: FocusDoc;
   317→   317→  if (modoCupom) {
   318→   318→    doc = 'nfce'; // BLOCO 108: modo cupom e SEMPRE NFC-e anonima (sem consumidor)
   319→   319→  } else if (tipoEscolhido === 'NFE' || tipoEscolhido === 'NFCE') {
   320→   320→    doc = tipoEscolhido === 'NFE' ? 'nfe' : 'nfce';
   321→   321→  } else {
   322→   322→    doc = cpfCnpj === 'CNPJ' ? 'nfe' : 'nfce';
   323→   323→  }
   324→   324→
   325→   325→  // ------------------------------------------------------------------
   326→   326→  // BLOCO 137 (regra do dono): NFC-e ANONIMA = PREÇO ESPECIAL. A NFC-e sai
   327→   327→  // sem consumidor (CONSUMIDOR NAO IDENTIFICADO) quando nao tem CPF
   328→   328→  // digitado E o cadastro nao tem documento valido — ou quando e o modo
   329→   329→  // cupom (BLOCO 108). Nesses casos os itens vao no PREÇO CUPOM do produto
   330→   330→  // (preco_cupom; 0 = custo; 0 = preço da venda) e o total da NFC e a soma
   331→   331→  // desses valores. BUG do papel corrigido: a NFC-e "sem CPF" emitida pela
   332→   332→  // tela 8 do PDV ou pela fila do admin saia no PREÇO CHEIO da venda porque
   333→   333→  // so o modo cupom explicito precificava (a mensagem do XML mostrava
   334→   334→  // "Venda ... registrada pelo app" em vez de "Venda cupom ...").

   338→   338→  const municipio = String(client.endereco_municipio || '').trim();

   349→   349→  // CNPJ (nem na tela ERP Fiscal, nem no env), recusamos ANTES de marcar a

   353→   353→      String(erp.emitente.cnpj || '').replace(/\D/g, '') || String(cfg.cnpj || '').replace(/\D/g, '');

   394→   394→      codigo: String(p.id || '').slice(0, 8),
   395→   395→      descricao: String(p.nome || 'Produto'),
   396→   396→      ncm: fiscalCols ? (p.ncm as string) || undefined : undefined,
   397→   397→      cfop: fiscalCols ? (p.cfop as string) || undefined : undefined,
   398→   398→      ean: fiscalCols ? (p.ean as string) || undefined : undefined,
   399→   399→      unidade: fiscalCols ? ((p.unidade as string) || 'UN') : 'UN',

   428→   428→    ? `Venda cupom ${vendaCurta} — NFC-e nao identificada${modoCupom ? ' (modo cupom)' : ''}. Emitida em ${carimbo()}.`

   430→   430→
   431→   431→  // Destinatario + payload por tipo de documento

   433→   433→  if (doc === 'nfe') {

   435→   435→    // emite no sistema externo dele) — aqui o app so CONTROLA a venda como

   441→   441→    if (!razao) faltando.push('Razao social / nome fantasia');

   443→   443→      return BAD(422, `Cadastro incompleto para NF-e. Complete no cadastro do cliente: ${faltando.join(', ')}.`);

   448→   448→      inscricao_estadual: String(client.inscricao_estadual || '').trim() || undefined,

   451→   451→      bairro: bairro || undefined,
   452→   452→      municipio: municipio || undefined,
   453→   453→      uf: uf || undefined,
   454→   454→      cep: String(client.endereco_cep || '').replace(/\D/g, '') || undefined,

   457→   457→    };
   458→   458→    payload = montarPayloadNfe({
   459→   459→      dataEmissao,
   460→   460→      natureza: cfg.natureza,
   461→   461→      ufEmitente: cfg.ufEmitente,
   462→   462→      destinatario,
   463→   463→      itens: payloadItems,
   464→   464→      valorTotalVenda: valorTotal,

   469→   469→  } else {
   470→   470→    // NFC-e (Bloco 12): escolha explicita do vendedor na tela de venda

   481→   481→        nome: (nomeConsumidor || 'CONSUMIDOR').slice(0, 60),
   482→   482→        cpf_cnpj: cpfNaNota,
   483→   483→        logradouro: logradouro || undefined,
   484→   484→        numero: numeroDoEndereco(client.endereco_numero as string, logradouro),
   485→   485→        bairro: bairro || undefined,
   486→   486→        municipio: municipio || undefined,
   487→   487→        uf: uf || cfg.ufEmitente,

   556→   556→  // NF-e), PDF e XML sao arquivados no NOSSO storage em background
