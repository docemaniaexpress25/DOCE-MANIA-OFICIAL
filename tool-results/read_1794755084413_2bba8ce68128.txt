   116→async function resolveEntregadorUuid(supabase: any, userId: string): Promise<string | null> {
   117→  if (!userId) return null;
   118→  if (RE_UUID.test(userId)) return userId;
   119→  const perfil = String(userId).toUpperCase();
   120→  try {
   121→    const { data, error } = await supabase
   122→      .from('app_users').select('id, perfil, ativo').eq('perfil', perfil).eq('ativo', true);
   123→    if (!error && Array.isArray(data)) {
   124→      if (data.length === 1) return data[0].id as string;
   125→      // 2+ usuarios reais: nao da para adivinhar quem esta entregando —
   126→      // mantem o UUID do acesso fixo (sem quebrar; o nominal volta quando
   127→      // entrarem com o login real de um deles).
   128→    }
   129→    // BLOCO 33: ids NAO-uuid (ex. mock E2E 'adminuser') — o toUpperCase do id
   130→    // nao e perfil. Tenta localizar o usuario pelo proprio id.
   131→    const alt = await supabase.from('app_users').select('id, ativo').eq('id', userId).eq('ativo', true).limit(1);
   132→    if (!alt.error && Array.isArray(alt.data) && alt.data.length === 1) return alt.data[0].id as string;
   133→  } catch { /* app_users inacessivel: segue com o UUID fixo */ }
   134→  return UUID_ACESSO_FIXO[perfil] || null;
   135→}

   214→ * venda ganha a taxa de PRONTA ENTREGA do produto (foi ele que vendeu e
   215→ * levou). Entregador dedicado/admin: taxa de PRE-VENDA, como sempre.

   218→async function registrarComissaoPreVenda(supabase: any, sale: any, quitada: boolean, usarComissaoPe = false) {
   219→  try {
   220→    // BLOCO 118 (regra do dono): comissao desativada para o David (R5) —
   221→    // nao calcula nem grava (o trigger do SQL 118 no banco tambem descarta).
   222→    if (sale.vendedor_id) {
   223→      const { data: vd118 } = await supabase
   224→        .from('app_users').select('nome').eq('id', sale.vendedor_id).limit(1).maybeSingle();
   225→      if (vd118 && /^david/i.test(String(vd118.nome || '').trim())) return;
   226→    }
   227→
   228→    const { data: items } = await supabase
   229→      .from('sale_items').select('produto_id, quantidade, preco_venda').eq('sale_id', sale.id);
   230→    const itens = items || [];
   231→    if (itens.length === 0) return;
   232→
   233→    const comissao = usarComissaoPe
   234→      ? await comissaoPeDeItens(supabase, itens)
   235→      : await comissaoPvDeItens(supabase, itens);
   236→    if (!(comissao > 0)) return;
   237→
   238→    const total = Math.max(Number(sale.valor_total || 0), 0.01);
   239→    const payload = {
   240→      seller_id: sale.vendedor_id,
   241→      valor_comissao: comissao,
   242→      valor_base: round2(Number(sale.valor_total || 0)),
   243→      percentual: round2((comissao / total) * 100),
   244→      status: quitada ? 'DISPONIVEL' : 'A_RECEBER',
   245→    };
   246→
   247→    // evita duplicar se a parada for reaberta e re-entregue: atualiza a row
   248→    // (a comissao conta no dia em que a entrega foi EFETIVADA: created_at
   249→    // anda junto com a re-entrega — data de entrega = data da comissao)
   250→    const { data: existente } = await supabase
   251→      .from('commissions').select('id').eq('sale_id', sale.id).limit(1);
   252→    if (existente && existente.length > 0) {
   253→      await supabase.from('commissions')
   254→        .update({ ...payload, created_at: new Date().toISOString() })
   255→        .eq('id', existente[0].id);
   256→    } else {
   257→      await supabase.from('commissions').insert({
   258→        sale_id: sale.id,
   259→        ...payload,
   260→        valor_pago: 0,
   261→        created_at: new Date().toISOString(),
   262→      });
   263→    }
   264→  } catch (e: any) {
   265→    // comissao nunca deve travar a entrega
   266→    console.error('[pre-venda] comissao pre-venda falhou:', e?.message);
   267→  }
   268→}

   324→async function enrichParadas(supabase: any, salesRows: any[]) {

   327→  const saleIds = salesRows.map((s: any) => s.id);
   328→

   330→    fetchClientsFlex(supabase, clientIds),
   331→    supabase.from('sale_items').select('sale_id, produto_id, quantidade, preco_venda').in('sale_id', saleIds),
   332→    supabase.from('entrega_eventos').select('sale_id, status, motivo, criado_em').in('sale_id', saleIds).order('criado_em', { ascending: true }),
   333→    // Bloco 6: foto do boleto (so consulta se a coluna existir)
   334→    hasBoletoFotoColumn().then(ok =>
   335→      ok ? supabase.from('entrega_eventos').select('sale_id').not('foto', 'is', null).in('sale_id', saleIds)
   336→         : Promise.resolve({ data: [] as any[] })),
   337→  ]);
   338→
   339→  const clientMap: Record<string, any> = {};
   340→  (clientsRes.data || []).forEach((c: any) => { clientMap[c.id] = c; });
   341→
   342→  // Nomes dos produtos
   343→  const prodIds = [...new Set((itemsRes.data || []).map((i: any) => i.produto_id))];
   344→  const prodMap: Record<string, string> = {};
   345→  if (prodIds.length > 0) {
   346→    const { data: prods } = await supabase.from('products').select('id, nome').in('id', prodIds);
   347→    (prods || []).forEach((p: any) => { prodMap[p.id] = p.nome; });
   348→  }
   349→
   350→  const itemsBySale: Record<string, any[]> = {};
   351→  (itemsRes.data || []).forEach((i: any) => {
   352→    (itemsBySale[i.sale_id] = itemsBySale[i.sale_id] || []).push({
   353→      nome: prodMap[i.produto_id] || 'Produto',
   354→      produtoId: i.produto_id,
   355→      quantidade: i.quantidade,
   356→      precoVenda: Number(i.preco_venda || 0),
   357→    });
   358→  });
   359→
   360→  const temFotoSet: Record<string, boolean> = {};
   361→  (fotosRes.data || []).forEach((f: any) => { temFotoSet[f.sale_id] = true; });
   362→
   363→  const eventsBySale: Record<string, any[]> = {};
   364→  (eventsRes.data || []).forEach((e: any) => {
   365→    (eventsBySale[e.sale_id] = eventsBySale[e.sale_id] || []).push({ status: e.status, motivo: e.motivo, criado_em: e.criado_em });
   366→  });

   520→  const { error } = await supabase.from('sales').update(payload).eq('id', saleId);
